\subsection*{Section 3: Return Diagnostics \& Higher Moments}

\begin{formulabox}[Daily Log Return and Annualized Volatility Scaling]
\begin{equation*}
R_t = \ln\left(\frac{P_t}{P_{t-1}}\right), \qquad \hat{\sigma} = \sqrt{\frac{1}{N-1}\sum_{t=1}^N (R_t - \hat{\mu})^2}, \qquad \sigma_{\text{annual}} = \hat{\sigma} \times \sqrt{252}
\end{equation*}
\textbf{Where:}
\begin{itemize}[leftmargin=1.8em, noitemsep]
    \item $R_t =$ Continuously compounded daily log return on trading session $t$
    \item $P_t, P_{t-1} =$ Adjusted closing price on day $t$ and day $t-1$
    \item $\hat{\mu} =$ Sample arithmetic mean daily return ($\frac{1}{N}\sum R_t$)
    \item $\hat{\sigma} =$ Daily standard deviation (sample volatility)
    \item $\sigma_{\text{annual}} =$ Annualized historical volatility scaled over $252$ trading days
    \item $N =$ Number of daily return observations in the historical sample
\end{itemize}
\end{formulabox}

\begin{formulabox}[Sample Skewness, Kurtosis, and Jarque--Bera Test Statistic]
\begin{equation*}
\hat{S} = \frac{\frac{1}{N}\sum (R_t - \hat{\mu})^3}{\hat{\sigma}^3}, \qquad \hat{K}_{\text{excess}} = \frac{\frac{1}{N}\sum (R_t - \hat{\mu})^4}{\hat{\sigma}^4} - 3, \qquad JB = \frac{N}{6}\left(\hat{S}^2 + \frac{\hat{K}_{\text{excess}}^2}{4}\right) \sim \chi^2(2)
\end{equation*}
\textbf{Where:}
\begin{itemize}[leftmargin=1.8em, noitemsep]
    \item $\hat{S} =$ Sample skewness measuring distribution asymmetry
    \item $\hat{K}_{\text{excess}} =$ Sample excess kurtosis measuring tail heaviness relative to Gaussian ($3$)
    \item $JB =$ Jarque--Bera test statistic for normality
    \item $\chi^2(2) =$ Chi-Square distribution with 2 degrees of freedom (Critical value $= 5.9915$ at $5\%$)
    \item \textbf{Decision Rule:} Reject Gaussian normality if $JB > 5.9915$
\end{itemize}
\end{formulabox}

\subsection*{Section 4: Value at Risk (VaR) \& Backtesting}

\begin{formulabox}[Parametric and Historical Simulation VaR Models]
\begin{equation*}
\VaR_{\text{param}, 1-\alpha}^{1\text{d}} = V_0 \times \left(Z_{\alpha} \cdot \sigma_{\text{daily}} - \mu_{\text{daily}}\right), \qquad \VaR_{\text{hist}, 1-\alpha}^{1\text{d}} = - V_0 \cdot Q_{\alpha}\left(\{R_t\}_{t=1}^N\right)
\end{equation*}
\begin{equation*}
\VaR_{\alpha, T_h} = \VaR_{\alpha, 1\text{d}} \times \sqrt{T_h}, \qquad \CVaR_{\alpha}^{1\text{d}} = - V_0 \cdot \frac{1}{\lfloor N \alpha \rfloor}\sum_{t=1}^{\lfloor N \alpha \rfloor} R_{(t)}
\end{equation*}
\textbf{Where:}
\begin{itemize}[leftmargin=1.8em, noitemsep]
    \item $\VaR_{\text{param}}, \VaR_{\text{hist}} =$ Parametric (Gaussian) and Non-Parametric Historical Simulation VaR
    \item $V_0 =$ Total portfolio capital commitment ($L \times S_0$)
    \item $Z_{\alpha} =$ Standard normal critical value ($1.645$ for $95\%$, $2.326$ for $99\%$)
    \item $Q_{\alpha} =$ Empirical $\alpha$-quantile of sorted historical returns
    \item $T_h =$ Multi-period holding horizon ($T_h = 10$ for regulatory benchmark)
    \item $\CVaR_{\alpha} =$ Expected Shortfall (Conditional VaR) measuring mean loss beyond cutoff
    \item $R_{(t)} =$ Ascending order statistics ($R_{(1)} \le R_{(2)} \le \dots \le R_{(N)}$)
\end{itemize}
\end{formulabox}

\begin{formulabox}[Kupiec Proportion of Failures (POF) Likelihood Ratio Statistic]
\begin{equation*}
\LR_{\POF} = -2 \ln \left[ \frac{(1 - p_0)^{N - x} p_0^x}{(1 - \hat{p})^{N - x} \hat{p}^x} \right] \sim \chi^2(1)
\end{equation*}
\textbf{Where:}
\begin{itemize}[leftmargin=1.8em, noitemsep]
    \item $\LR_{\POF} =$ Likelihood ratio test statistic for model failure frequency
    \item $N =$ Number of trading days in backtest sample
    \item $x =$ Count of observed VaR exception violations ($\sum \mathbb{I}_{\{R_t < -\VaR\}}$)
    \item $p_0 =$ Nominal failure rate ($0.05$ for $95\%$, $0.01$ for $99\%$)
    \item $\hat{p} =$ Sample failure rate ($x / N$)
    \item \textbf{Decision Rule:} Reject model if $\LR_{\POF} > 3.8415$ ($\chi^2_{0.05, 1}$)
\end{itemize}
\end{formulabox}

\subsection*{Section 5: Option Valuation via Black--Scholes--Merton}

\begin{formulabox}[Merton's Continuous Dividend-Adjusted Option Pricing Formulas]
\begin{align*}
C_{\text{merton}} &= S_0 e^{-q T} N(d_1) - K e^{-r T} N(d_2) \\
P_{\text{merton}} &= K e^{-r T} N(-d_2) - S_0 e^{-q T} N(-d_1) \\
d_1 = \frac{\ln(S_0 / K) + \left(r - q + \frac{\sigma^2}{2}\right)T}{\sigma \sqrt{T}}&, \qquad d_2 = d_1 - \sigma \sqrt{T}
\end{align*}
\textbf{Where:}
\begin{itemize}[leftmargin=1.8em, noitemsep]
    \item $C_{\text{merton}}, P_{\text{merton}} =$ Theoretical dividend-adjusted European call and put option prices
    \item $S_0 =$ Underlying spot price of the stock
    \item $K =$ Strike price of the option contract
    \item $T =$ Time to expiration in years ($\text{Days to expiry} / 365$)
    \item $r =$ Continuous risk-free interest rate (annualized)
    \item $q =$ Continuous annual dividend yield ($\text{Annual Payout} / S_0$)
    \item $\sigma =$ Annualized volatility of underlying returns
    \item $N(\cdot) =$ Cumulative standard normal distribution function
\end{itemize}
\end{formulabox}

\begin{formulabox}[Analytical Dividend-Adjusted Option Greeks]
\begin{align*}
\Delta_{\text{call}} &= e^{-q T} N(d_1), \qquad \Delta_{\text{put}} = -e^{-q T} N(-d_1) \\
\Gamma &= \frac{e^{-q T} N'(d_1)}{S_0 \sigma \sqrt{T}}, \qquad \mathcal{V} = \frac{S_0 e^{-q T} \sqrt{T} N'(d_1)}{100}
\end{align*}
\textbf{Where:}
\begin{itemize}[leftmargin=1.8em, noitemsep]
    \item $\Delta =$ Option Delta measuring directional sensitivity ($\partial V / \partial S$)
    \item $\Gamma =$ Option Gamma measuring rate of change of Delta ($\partial^2 V / \partial S^2$)
    \item $\mathcal{V} =$ Option Vega measuring price sensitivity to $1\%$ change in volatility ($\partial V / \partial \sigma$)
    \item $N'(x) =$ Standard normal probability density function $\frac{1}{\sqrt{2\pi}} e^{-x^2/2}$
\end{itemize}
\end{formulabox}

\subsection*{Section 6: Integrated Derivative Hedging Strategies}

\begin{formulabox}[Protective Put and Zero-Cost Collar Payoff Functions]
\begin{align*}
\Pi_{\text{prot}}(S_T) &= (S_T - S_0) + \max(K - S_T, 0) - P(K) \\
\Pi_{\text{collar}}(S_T) &= (S_T - S_0) + \max(K_1 - S_T, 0) - \max(S_T - K_2, 0) - [P(K_1) - C(K_2)]
\end{align*}
\begin{equation*}
\text{VaR Reduction (\%)} = \left[ \frac{\mathcal{L}_{\text{unhedged}}(\VaR) - \mathcal{L}_{\text{hedged}}(\VaR)}{\mathcal{L}_{\text{unhedged}}(\VaR)} \right] \times 100\%
\end{equation*}
\textbf{Where:}
\begin{itemize}[leftmargin=1.8em, noitemsep]
    \item $\Pi_{\text{prot}}(S_T) =$ Net profit or loss per share of protective put at expiry $T$
    \item $\Pi_{\text{collar}}(S_T) =$ Net profit or loss per share of collar structure at expiry $T$
    \item $K, K_1 =$ Strike prices of purchased protective put options ($K \le S_0$)
    \item $K_2 =$ Strike price of written covered call option ($K_2 \ge S_0$)
    \item $P(K), C(K_2) =$ Upfront market option premiums for put and call
    \item $\mathcal{L}_{\text{unhedged}}(\VaR) =$ Capital loss of unhedged equity lot at the $99\%$ 10-day VaR horizon
    \item $\mathcal{L}_{\text{hedged}}(\VaR) =$ Net capital loss of hedged portfolio at that exact same VaR spot price
    \item $\text{VaR Reduction (\%)} =$ Percentage of extreme tail loss insulated by derivative positioning
\end{itemize}
\end{formulabox}
