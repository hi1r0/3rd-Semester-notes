% 1. Decision Rule Box (Crisp greyscale callout for anomaly criteria)
\newtcolorbox{decisionrule}[1][]{
    enhanced,
    colback=boxfill,
    colframe=pureblack,
    title=#1,
    fonttitle=\bfseries\footnotesize\color{white},
    coltitle=white,
    attach boxed title to top left={yshift=-1.5mm, xshift=3mm},
    boxed title style={colback=pureblack, arc=0.8mm},
    boxrule=0.8pt,
    arc=1.2mm,
    top=2.5mm,
    bottom=2mm,
    left=3mm,
    right=3mm,
    before skip=3pt,
    after skip=3pt,
    breakable
}
