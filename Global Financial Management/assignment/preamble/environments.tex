% 1. Definition Box
\newtcbtheorem[number within=section]{definition}{Definition}{
    enhanced,
    colback=boxfill,
    colframe=pureblack,
    coltitle=white,
    fonttitle=\bfseries\sffamily\footnotesize,
    attach boxed title to top left={yshift=-1.5mm, xshift=4mm},
    boxed title style={colback=pureblack, arc=0.8mm},
    minipage boxed title=\linewidth-1.5cm,
    separator sign={: },
    boxrule=0.8pt,
    arc=1.2mm,
    top=2.5mm,
    bottom=2mm,
    left=3mm,
    right=3mm,
    before skip=4pt,
    after skip=4pt,
    breakable
}{def}
