\subsection*{Section 1 \& 2: FDI Accounting \& Capital Stock Accumulation}

\begin{formulabox}[Direct Investment Balance of Payments Capital Accumulation Identity]
\begin{equation}
    \text{Stock}_t = \text{Stock}_{t-1} + \text{Flow}_t + \Delta \text{Valuation}_t^{\text{FX}} + \Delta \text{Valuation}_t^{\text{Price}} + \text{Other Adjustments}_t
\end{equation}
\textbf{Where:}
\begin{itemize}[leftmargin=1.2em, itemsep=1.5pt]
    \item $\text{Stock}_t$ = Total recorded FDI capital stock position at the close of calendar year $t$.
    \item $\text{Stock}_{t-1}$ = FDI capital stock position at the close of preceding year $t-1$.
    \item $\text{Flow}_t$ = Net direct investment transactions (financial inflows minus outflows) during year $t$.
    \item $\Delta \text{Valuation}_t^{\text{FX}}$ = Currency translation revaluation resulting from foreign exchange rate fluctuations.
    \item $\Delta \text{Valuation}_t^{\text{Price}}$ = Asset price revaluation resulting from equity market shifts and asset write-downs.
    \item $\text{Other Adjustments}_t$ = Statistical reclassifications, perimeter coverage changes, and accounting revisions.
\end{itemize}
\end{formulabox}

\begin{formulabox}[Net Direct Investment Position (Capital Export Balance)]
\begin{equation}
    \text{NDIP}_t = \text{Stock}_t^{\text{Outward}} - \text{Stock}_t^{\text{Inward}}
\end{equation}
\textbf{Where:}
\begin{itemize}[leftmargin=1.2em, itemsep=1.5pt]
    \item $\text{NDIP}_t$ = Net Direct Investment Position of the domestic economy at period $t$.
    \item $\text{Stock}_t^{\text{Outward}}$ = Total outward FDI capital stock deployed abroad by domestic residents.
    \item $\text{Stock}_t^{\text{Inward}}$ = Total inward FDI capital stock held by foreign entities domestically.
    \item Condition: $\text{NDIP}_t > 0 \iff$ Net Capital Exporter (Structural Global Creditor Economy).
    \item Condition: $\text{NDIP}_t < 0 \iff$ Net Capital Importer (Structural Global Debtor Economy).
\end{itemize}
\end{formulabox}

\subsection*{Section 3: Macroeconomic Integration Ratios \& Growth Metrics}

\begin{formulabox}[Direct Investment Macroeconomic Intensity Ratios]
\begin{equation}
    \Phi_t^{\text{Inward}} = \frac{\text{Stock}_t^{\text{Inward}}}{\text{GDP}_t}, \qquad \Phi_t^{\text{Outward}} = \frac{\text{Stock}_t^{\text{Outward}}}{\text{GDP}_t}
\end{equation}
\textbf{Where:}
\begin{itemize}[leftmargin=1.2em, itemsep=1.5pt]
    \item $\Phi_t^{\text{Inward}}$ = Inward direct investment penetration intensity relative to domestic economic output.
    \item $\Phi_t^{\text{Outward}}$ = Outward direct investment propensity intensity relative to domestic economic output.
    \item $\text{GDP}_t$ = Nominal Gross Domestic Product in domestic currency at period $t$.
    \item $\text{Stock}_t^{\text{Inward}}, \text{Stock}_t^{\text{Outward}}$ = Total inward and outward direct investment capital stocks at period $t$.
\end{itemize}
\end{formulabox}

\begin{formulabox}[Compound Annual Growth Rate (CAGR) Formulation]
\begin{equation}
    \text{CAGR}_{t_0, t_n} = \left( \frac{\text{Stock}_{t_n}}{\text{Stock}_{t_0}} \right)^{\frac{1}{t_n - t_0}} - 1
\end{equation}
\textbf{Where:}
\begin{itemize}[leftmargin=1.2em, itemsep=1.5pt]
    \item $\text{CAGR}_{t_0, t_n}$ = Geometric mean annual growth rate of capital stock over $n$ calendar periods.
    \item $\text{Stock}_{t_0}$ = Direct investment capital stock at baseline period $t_0$.
    \item $\text{Stock}_{t_n}$ = Direct investment capital stock at terminal period $t_n$.
    \item $t_n - t_0$ = Total elapsed duration in years between baseline and terminal observations.
\end{itemize}
\end{formulabox}

\subsection*{Section 5: Conduit FDI \& Intermediation Distortion Indices}

\begin{formulabox}[Conduit Intermediation Distortion Index (Phantom FDI Ratio)]
\begin{equation}
    \Theta_i = \frac{\text{Stock}_{i}^{\text{Immediate}}}{\text{Stock}_{i}^{\text{UBO}}}
\end{equation}
\textbf{Where:}
\begin{itemize}[leftmargin=1.2em, itemsep=1.5pt]
    \item $\Theta_i$ = Conduit Intermediation Distortion Ratio for bilateral partner jurisdiction $i$.
    \item $\text{Stock}_{i}^{\text{Immediate}}$ = Direct investment capital stock attributed under the immediate creditor principle.
    \item $\text{Stock}_{i}^{\text{UBO}}$ = Direct investment capital stock attributed under the Ultimate Beneficial Owner principle.
    \item Condition: $\Theta_i > 1 \iff$ Conduit / Pass-Through Jurisdiction (Intermediary Capital Routing).
    \item Condition: $\Theta_i < 1 \iff$ Indirect Investor Jurisdiction (Ultimate Capital Origin Masked via Intermediaries).
    \item Condition: $\Theta_i = 1 \iff$ Neutral Jurisdiction (Pure Direct Investment with Zero Conduit Intermediation).
\end{itemize}
\end{formulabox}

\subsection*{Section 6 \& 8: Multinational Operational Multipliers \& Spillovers}

\begin{formulabox}[Multinational Enterprise Global Labor Multiplier]
\begin{equation}
    \Lambda_t^{\text{Labor}} = \frac{L_t^{\text{Abroad}}}{L_t^{\text{Parent}}}
\end{equation}
\textbf{Where:}
\begin{itemize}[leftmargin=1.2em, itemsep=1.5pt]
    \item $\Lambda_t^{\text{Labor}}$ = Multinational enterprise global overseas-to-domestic labor multiplier.
    \item $L_t^{\text{Abroad}}$ = Aggregate personnel employed across foreign non-resident subsidiaries of domestic MNEs.
    \item $L_t^{\text{Parent}}$ = Total domestic personnel employed by multinational parent enterprises resident in the home country.
    \item Condition: $\Lambda_t^{\text{Labor}} > 1 \iff$ Outward-Oriented Multinational Operational Production Footprint.
\end{itemize}
\end{formulabox}

\begin{formulabox}[Gross Fixed Capital Formation (GFCF) Investment Elasticity of FDI]
\begin{equation}
    \varepsilon_{\text{GFCF}} = \frac{\Delta \text{GFCF}_t / \text{GFCF}_{t-1}}{\Delta \text{FDI}_t / \text{FDI}_{t-1}}
\end{equation}
\textbf{Where:}
\begin{itemize}[leftmargin=1.2em, itemsep=1.5pt]
    \item $\varepsilon_{\text{GFCF}}$ = Elasticity of domestic Gross Fixed Capital Formation with respect to direct investment.
    \item $\Delta \text{GFCF}_t$ = Incremental change in annual domestic gross fixed capital expenditures in period $t$.
    \item $\text{GFCF}_{t-1}$ = Gross Fixed Capital Formation recorded in preceding baseline period $t-1$.
    \item $\Delta \text{FDI}_t$ = Incremental change in gross foreign direct investment capital expenditures in period $t$.
    \item $\text{FDI}_{t-1}$ = Gross foreign direct investment expenditures recorded in preceding period $t-1$.
\end{itemize}
\end{formulabox}
