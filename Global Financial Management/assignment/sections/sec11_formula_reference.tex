\begin{formulabox}[Direct Investment BOP Accumulation Identity \& Net Investment Position]
\footnotesize
\begin{align}
    \text{Stock}_t &= \text{Stock}_{t-1} + \text{Flow}_t + \Delta \text{Valuation}_t^{\text{FX}} + \Delta \text{Valuation}_t^{\text{Price}} + \text{Other Adjustments}_t \\
    \text{NDIP}_t &= \text{Stock}_t^{\text{Outward}} - \text{Stock}_t^{\text{Inward}} \qquad (\text{NDIP}_t > 0 \iff \text{Structural Net Capital Exporter})
\end{align}
\textbf{Where:}
\begin{itemize}[leftmargin=1.2em, itemsep=1pt]
    \item $\text{Stock}_t, \text{Stock}_{t-1}$ = Recorded FDI capital stock at period $t$ and preceding year $t-1$.
    \item $\text{Flow}_t$ = Net cross-border financial transactions (inflows minus outflows) recorded in BOP.
    \item $\Delta \text{Valuation}_t^{\text{FX}}, \Delta \text{Valuation}_t^{\text{Price}}$ = Foreign exchange translation and asset price revaluations.
    \item $\text{NDIP}_t$ = Net Direct Investment Position; Condition: $\text{NDIP}_t > 0 \iff$ Net Capital Exporter.
\end{itemize}
\end{formulabox}

\begin{formulabox}[Macroeconomic FDI Penetration Intensity \& Compound Annual Growth]
\footnotesize
\begin{equation}
    \Phi_t^{\text{Inward}} = \frac{\text{Stock}_t^{\text{Inward}}}{\text{GDP}_t}, \qquad \Phi_t^{\text{Outward}} = \frac{\text{Stock}_t^{\text{Outward}}}{\text{GDP}_t}, \qquad \text{CAGR} = \left( \frac{\text{Stock}_{t_n}}{\text{Stock}_{t_0}} \right)^{\frac{1}{t_n - t_0}} - 1
\end{equation}
\textbf{Where:}
\begin{itemize}[leftmargin=1.2em, itemsep=1pt]
    \item $\Phi_t^{\text{Inward}}, \Phi_t^{\text{Outward}}$ = Inward and outward FDI penetration intensity relative to domestic GDP.
    \item $\text{CAGR}$ = Geometric compound annual growth rate of capital stock over $t_n - t_0$ elapsed years.
\end{itemize}
\end{formulabox}

\begin{formulabox}[Conduit Intermediation Distortion Index (Phantom FDI Ratio)]
\footnotesize
\begin{equation}
    \Theta_i = \frac{\text{Stock}_{i}^{\text{Immediate}}}{\text{Stock}_{i}^{\text{UBO}}} \qquad (\Theta_i > 1 \iff \text{Conduit}; \; \Theta_i < 1 \iff \text{Masked Origin})
\end{equation}
\textbf{Where:}
\begin{itemize}[leftmargin=1.2em, itemsep=1pt]
    \item $\Theta_i$ = Conduit distortion ratio; $\text{Stock}_{i}^{\text{Immediate}}$ = Stock by immediate creditor; $\text{Stock}_{i}^{\text{UBO}}$ = Stock by UBO.
    \item $\Theta_i > 1 \iff$ Conduit Pass-Through Jurisdiction; $\Theta_i < 1 \iff$ Masked Origin Jurisdiction.
\end{itemize}
\end{formulabox}

\begin{formulabox}[Multinational Labor Multiplier \& Gross Fixed Capital Formation Elasticity]
\footnotesize
\begin{equation}
    \Lambda_t^{\text{Labor}} = \frac{L_t^{\text{Abroad}}}{L_t^{\text{Parent}}} \quad (\Lambda_t^{\text{Labor}} > 1), \qquad \varepsilon_{\text{GFCF}} = \frac{\Delta \text{GFCF}_t / \text{GFCF}_{t-1}}{\Delta \text{FDI}_t / \text{FDI}_{t-1}}
\end{equation}
\textbf{Where:}
\begin{itemize}[leftmargin=1.2em, itemsep=1pt]
    \item $\Lambda_t^{\text{Labor}}$ = Multinational overseas-to-domestic labor multiplier ($L^{\text{Abroad}} / L^{\text{Parent}}$).
    \item $\varepsilon_{\text{GFCF}}$ = Elasticity of domestic Gross Fixed Capital Formation ($\text{GFCF}$) to direct investment ($\text{FDI}$).
\end{itemize}
\end{formulabox}
