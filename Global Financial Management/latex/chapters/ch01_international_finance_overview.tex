% ------------------------------------------------------------------------------
\subsection{1. IBRD (International Bank for Reconstruction and Development)}
\begin{formulabox}[Institutional Profile: IBRD]
\begin{itemize}[leftmargin=*, noitemsep]
    \item World's largest Development Bank.
    \item \textbf{Membership:} 189 countries member in this organisation.
\end{itemize}
\end{formulabox}

\noindent\textbf{Mission:}
\begin{itemize}[leftmargin=*]
    \item Providing loans.
    \item Guarantee, Risk management products, advisory services for middle income and credit worthy and low income countries.
    \item Coordination responses to regional and global challenges.
\end{itemize}

% ------------------------------------------------------------------------------
\subsection{2. IDA (International Development Association)}
\begin{formulabox}[Institutional Profile: IDA]
\begin{itemize}[leftmargin=*, noitemsep]
    \item \textbf{Established:} Formed in 1960.
    \item \textbf{Purpose:} Development assistance and poverty reduction.
    \item \textbf{Membership:} 174 Member of countries.
\end{itemize}
\end{formulabox}

% ------------------------------------------------------------------------------
\subsection{3. IFC (International Finance Corporation)}
\begin{formulabox}[Institutional Profile: IFC]
\begin{itemize}[leftmargin=*, noitemsep]
    \item \textbf{Established:} Established on 1956.
\end{itemize}
\end{formulabox}

\noindent\textbf{Mission:}
\begin{itemize}[leftmargin=*]
    \item To end extreme poverty.
    \item To encourage private sector growth in developing countries.
\end{itemize}

\noindent\textbf{Objective:}
\begin{itemize}[leftmargin=*]
    \item \textbf{Investment money:} Provide loans and equity investment.
    \item \textbf{Mobilize capital:} Through commercial bank.
    \item \textbf{Provide advice:} Related to better policies, high environmental and social standards.
\end{itemize}

% ------------------------------------------------------------------------------
\subsection{4. MIGA (Multilateral Investment Guarantee Agency)}
\begin{formulabox}[Institutional Profile: MIGA]
\begin{itemize}[leftmargin=*, noitemsep]
    \item \textbf{Established:} Established in 1988.
    \item \textbf{Membership:} 29 countries are member in this organisation.
\end{itemize}
\end{formulabox}

\noindent\textbf{Motives:}
\begin{itemize}[leftmargin=*]
    \item MIGA covers political risk like war.
\end{itemize}

\noindent\textbf{Objective:}
\begin{itemize}[leftmargin=*]
    \item Support economic growth by encouraging FDI.
    \item Provides advisory services to help the developing countries.
\end{itemize}

% ------------------------------------------------------------------------------
\subsection{5. ICSID (International Centre for Settlement of Investment Disputes)}
\begin{formulabox}[Institutional Profile: ICSID]
\begin{itemize}[leftmargin=*, noitemsep]
    \item \textbf{Established:} Established on 1966.
    \item \textbf{Membership:} It had 143 Member countries.
    \item \textbf{Disputes:} Total disputes registered on 2021, almost 159 investment related disputes.
\end{itemize}
\end{formulabox}
