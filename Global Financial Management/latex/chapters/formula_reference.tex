\begin{formulabox}[Balance of Payments Equilibrium Identity]
\begin{equation*}
    \text{BOP} = \text{Current Account} + \text{Capital Account} + \text{Financial Account} + \text{Errors \& Omissions} = 0
\end{equation*}
\textbf{Where:}
\begin{itemize}[leftmargin=*, noitemsep]
    \item \textbf{Current Account:} Net balance on visible merchandise trade, invisible services, primary investment income, and unilateral transfers.
    \item \textbf{Capital Account:} Net cross-border capital transfers, debt forgiveness, and acquisition/disposal of non-produced, non-financial assets.
    \item \textbf{Financial Account:} Net foreign direct investment (FDI), foreign portfolio investment (FPI), commercial loans, and official reserve transactions.
    \item \textbf{Errors \& Omissions:} Statistical balancing item ensuring double-entry equality.
\end{itemize}
\end{formulabox}

\begin{formulabox}[SDR Multi-Currency Basket Valuation]
\begin{equation*}
    \text{SDR Value in USD} = \sum_{i=1}^{5} \left( q_i \times S_i \right)
\end{equation*}
\textbf{Where:}
\begin{itemize}[leftmargin=*, noitemsep]
    \item $q_i$: Fixed units of currency $i$ defined in the 5-year IMF SDR valuation basket.
    \item $S_i$: Spot market exchange rate of currency $i$ expressed in US Dollars.
    \item \textbf{Relative Basket Weights:} USD ($43.39\%$), EUR ($29.31\%$), CNY ($12.28\%$), JPY ($7.59\%$), GBP ($7.44\%$).
\end{itemize}
\end{formulabox}

\section*{Chapter 2: The Foreign Exchange Market}

\begin{formulabox}[Bid-Ask Spread and Percentage Spread]
\begin{align*}
    \text{Spread} &= \text{Ask Price} - \text{Bid Price} \\[0.3em]
    \text{Percentage Spread (\%)} &= \left[ \frac{\text{Ask Price} - \text{Bid Price}}{\text{Ask Price}} \right] \times 100
\end{align*}
\textbf{Where:}
\begin{itemize}[leftmargin=*, noitemsep]
    \item \textbf{Ask Price (Offer):} Exchange rate at which the quoting market maker sells foreign currency.
    \item \textbf{Bid Price:} Exchange rate at which the quoting market maker buys foreign currency.
\end{itemize}
\end{formulabox}

\begin{formulabox}[Covered Interest Rate Parity (Exact Forward Rate)]
\begin{equation*}
    F = S \times \left[ \frac{1 + r_h \times \frac{d}{365}}{1 + r_f \times \frac{d}{365}} \right]
\end{equation*}
\textbf{Where:}
\begin{itemize}[leftmargin=*, noitemsep]
    \item $F$: Outright forward exchange rate (Home currency per unit of foreign currency).
    \item $S$: Spot exchange rate (Home currency per unit of foreign currency).
    \item $r_h$: Annualized risk-free interest rate in the domestic (home) country.
    \item $r_f$: Annualized risk-free interest rate in the foreign country.
    \item $d$: Maturity duration of the forward contract in days.
\end{itemize}
\end{formulabox}

\begin{formulabox}[Linear Forward Approximation \& Annualized Premium/Discount]
\begin{align*}
    F_{\text{linear}} &\approx S \times \left[ 1 + (r_h - r_f) \times \frac{d}{365} \right] \\[0.3em]
    \text{Annualized Premium / Discount (\%)} &= \left[ \frac{F - S}{S} \right] \times \frac{365}{d} \times 100
\end{align*}
\textbf{Where:}
\begin{itemize}[leftmargin=*, noitemsep]
    \item $F$: Forward exchange rate; $S$: Spot exchange rate.
    \item $r_h - r_f$: Nominal interest rate differential between home and foreign country.
    \item $d$: Forward contract duration in days.
    \item \textbf{Condition:} $F > S \implies \text{Forward Premium}$; $F < S \implies \text{Forward Discount}$.
\end{itemize}
\end{formulabox}

\begin{formulabox}[Merchant TT Buying Rate (Clean Remittances)]
\begin{equation*}
    \text{TT Buying Rate} = \text{Interbank Spot Buying Rate (Bid)} - \text{Exchange Margin}
\end{equation*}
\textbf{Where:}
\begin{itemize}[leftmargin=*, noitemsep]
    \item \textbf{Interbank Spot Buying Rate:} Market bid price utilized as base by authorized dealer.
    \item \textbf{Exchange Margin:} Merchant loading percentage deducted to yield the customer buying quote.
    \item \textbf{FEDAI Rounding Convention:} Quoted rate rounded to the nearest multiple of $0.0025$.
\end{itemize}
\end{formulabox}

\begin{formulabox}[Merchant Bill Buying Rate (Commercial Trade Bills)]
\begin{equation*}
    \text{Bill Buying Rate} = \text{Forward Buying Rate (for Transit)} - \text{Exchange Margin}
\end{equation*}
\begin{equation*}
    \text{Forward Buying Rate} = \begin{cases} 
        \text{Spot Bid} - \text{Forward Discount}, & \text{if at Discount} \\
        \text{Spot Bid}, & \text{if at Premium}
    \end{cases}
\end{equation*}
\textbf{Where:}
\begin{itemize}[leftmargin=*, noitemsep]
    \item \textbf{Forward Discount:} Forward discount margin deducted for the normal transit period.
    \item \textbf{Exchange Margin:} Percentage margin deducted from the base buying rate.
    \item \textbf{FEDAI Rounding Convention:} Quoted rate rounded to the nearest multiple of $0.0025$.
\end{itemize}
\end{formulabox}

\begin{formulabox}[Recovery of Interest on Commercial Bills Purchased]
\begin{equation*}
    \text{Interest on Bill} = \text{Rupee Amount} \times r \times \left[ \frac{\text{NTP} + \text{Usance}}{365} \right]
\end{equation*}
\textbf{Where:}
\begin{itemize}[leftmargin=*, noitemsep]
    \item \textbf{Rupee Amount:} Gross rupee value credited to exporter at the Bill Buying Rate.
    \item $r$: Annual commercial bank lending / discount rate charged to customer.
    \item $\text{NTP}$: Normal Transit Period in days prescribed by FEDAI rules.
    \item $\text{Usance}$: Commercial credit tenor specified on the bill of exchange in days.
\end{itemize}
\end{formulabox}

\begin{formulabox}[Merchant TT Selling Rate (Outward Remittances)]
\begin{equation*}
    \text{TT Selling Rate} = \text{Interbank Spot Selling Rate (Ask)} + \text{Exchange Margin}
\end{equation*}
\textbf{Where:}
\begin{itemize}[leftmargin=*, noitemsep]
    \item \textbf{Interbank Spot Selling Rate:} Market ask / offer rate at which bank covers in interbank tier.
    \item \textbf{Exchange Margin:} Merchant loading percentage added to base selling rate.
    \item \textbf{FEDAI Rounding Convention:} Quoted rate rounded to the nearest multiple of $0.0025$.
\end{itemize}
\end{formulabox}

\begin{formulabox}[Cross-Rate Merchant TT Buying Rate]
\begin{equation*}
    \text{TT Buying Rate}_{\text{FC}} = \frac{\text{Interbank Spot Buying Rate (Bid)}_{\text{USD/INR}} - \text{Exchange Margin}}{\text{Overseas Market Spot Selling Rate (Ask)}_{\text{USD/FC}}}
\end{equation*}
\textbf{Where:}
\begin{itemize}[leftmargin=*, noitemsep]
    \item $\text{Interbank Spot Buying Rate (Bid)}_{\text{USD/INR}}$: Quoting dealer's local base buying price for US Dollars.
    \item $\text{Exchange Margin}$: Loading percentage deducted from the domestic base rate.
    \item $\text{Overseas Market Spot Selling Rate (Ask)}_{\text{USD/FC}}$: Interbank offer rate for foreign currency in overseas market (Singapore, Tokyo, London).
    \item $\text{Rounding Convention}$: Rate rounded off to nearest multiple of $0.0025$.
\end{itemize}
\end{formulabox}

\begin{formulabox}[Financial Outcome on Foreign Demand Draft Cancellation]
\begin{align*}
    \text{Net Financial Outcome} &= \text{Cancellation Proceeds} - \text{Original Cost} \\
    &= \text{FC Amount} \times \left( \text{TT Buying Rate} - \text{Original Issue Rate} \right)
\end{align*}
\textbf{Where:}
\begin{itemize}[leftmargin=*, noitemsep]
    \item $\text{TT Buying Rate}$: Net cross-rate applied by the bank to repurchase the cancelled draft.
    \item $\text{Original Issue Rate}$: Exchange rate paid by the customer when the draft was initially issued.
    \item $\text{Condition: } \text{Outcome} > 0 \implies \text{Customer Realizes Exchange Gain}$.
    \item $\text{Condition: } \text{Outcome} < 0 \implies \text{Customer Incurs Exchange Loss}$.
\end{itemize}
\end{formulabox}

\begin{formulabox}[Forward Delivery Horizon under FEDAI Rule 7]
\begin{align*}
    \text{Spot Baseline Date} &= \text{Trade Booking Date } (T) + 2 \text{ Business Days} \\[0.3em]
    \text{Fixed Maturity Date} &= \text{Spot Baseline Date} + \text{Contract Tenor} \\[0.3em]
    \text{Option Delivery Period} &\le 1 \text{ Calendar Month}
\end{align*}
\textbf{Where:}
\begin{itemize}[leftmargin=*, noitemsep]
    \item $T$: Transaction booking date.
    \item $\text{Spot Baseline Date}$: Inception anchor ($T+2$) from which forward duration commences.
    \item $\text{Contract Tenor}$: Duration of forward cover (e.g., 1 month, 2 months, 3 months).
    \item $\text{Option Delivery Period}$: Flexible window within which customer may deliver foreign exchange.
\end{itemize}
\end{formulabox}

\begin{formulabox}[Forward Merchant Bill Buying Rate with Transit and Usance Tenors]
\begin{equation*}
    \text{Forward Bill Buying Rate} = \left( \Spot_{\text{Bid}} \pm \text{Swap Points}_{M} \right) - \left[ \left( \Spot_{\text{Bid}} \pm \text{Swap Points}_{M} \right) \times m \right]
\end{equation*}
\textbf{Where:}
\begin{itemize}[leftmargin=*, noitemsep]
    \item $\Spot_{\text{Bid}}$: Interbank spot buying (bid) base rate for US Dollars.
    \item $\text{Swap Points}_{M}$: Monthly forward margin where tenor $M = \text{Forward Period} + \text{Transit Period} + \text{Usance Period}$.
    \item $\text{Ascending Margins (Premium)}$: Add premium; fractional duration rounded down to lower whole month ($M \leftarrow \lfloor M \rfloor$).
    \item $\text{Descending Margins (Discount)}$: Deduct discount; fractional duration rounded up to higher whole month ($M \leftarrow \lceil M \rceil$).
    \item $m$: Statutory exchange margin percentage (e.g., $0.10\%$).
    \item $\text{Rounding Rule}$: Final merchant quote rounded to nearest multiple of $0.0025$.
\end{itemize}
\end{formulabox}

\begin{formulabox}[FEDAI Option Forward Delivery Margin Selection Rule]
\begin{align*}
    &\textbf{Bank Purchases FX (Merchant Sells):} \\
    &\quad \begin{cases}
        \text{FC at Premium} \implies \text{Earliest Option Date } (\text{Lowest Premium Paid}) \\
        \text{FC at Discount} \implies \text{Latest Option Date } (\text{Highest Discount Deducted})
    \end{cases} \\[0.4em]
    &\textbf{Bank Sells FX (Merchant Buys):} \\
    &\quad \begin{cases}
        \text{FC at Premium} \implies \text{Latest Option Date } (\text{Highest Premium Charged}) \\
        \text{FC at Discount} \implies \text{Earliest Option Date } (\text{Lowest Discount Allowed})
    \end{cases}
\end{align*}
\textbf{Where:}
\begin{itemize}[leftmargin=*, noitemsep]
    \item $\text{Option Delivery Window}$: Contractual execution window not exceeding $1 \text{ calendar month}$.
    \item $\text{Principle of Quotation}$: Quoting dealer prices assuming merchant exercises option on date least favorable to bank.
    \item $\text{Ascending Points}$: Indicates foreign currency forward premium.
    \item $\text{Descending Points}$: Indicates foreign currency forward discount.
\end{itemize}
\end{formulabox}

\begin{formulabox}[Cross-Rate Forward Merchant Conversion Identity]
\begin{align*}
    \Forward_{\text{Buying (FC)}} &= \frac{\Forward_{\text{Domestic Interbank Buying (USD/INR)}}}{\Forward_{\text{Overseas Interbank Selling (USD/FC)}}} \\[0.5em]
    \Forward_{\text{Selling (FC)}} &= \frac{\Forward_{\text{Domestic Interbank Selling (USD/INR)}}}{\Forward_{\text{Overseas Interbank Buying (USD/FC)}}}
\end{align*}
\textbf{Where:}
\begin{itemize}[leftmargin=*, noitemsep]
    \item $\Forward_{\text{Domestic Interbank Buying (USD/INR)}}$: Mumbai forward dollar buying rate (Spot Bid $\pm$ forward margin less margin).
    \item $\Forward_{\text{Overseas Interbank Selling (USD/FC)}}$: Overseas market forward dollar selling rate (Spot Ask $\pm$ overseas forward margin).
    \item $\Forward_{\text{Domestic Interbank Selling (USD/INR)}}$: Mumbai forward dollar selling rate (Spot Ask $\pm$ forward margin plus margin).
    \item $\Forward_{\text{Overseas Interbank Buying (USD/FC)}}$: Overseas market forward dollar buying rate (Spot Bid $\pm$ overseas forward margin).
    \item $\text{Rounding Convention}$: Final synthesized cross-rate rounded to nearest multiple of $0.0025$.
\end{itemize}
\end{formulabox}


\begin{formulabox}[Relative Purchasing Power Parity (PPP) \& Multi-Period Compounding]
\begin{align*}
    \Spot_t &= \Spot_0 \left( \frac{1 + i_L}{1 + i_F} \right)^t \\[0.5em]
    \Spot_t &= \Spot_0 \prod_{k=1}^t \left( \frac{1 + i_{L,k}}{1 + i_{F,k}} \right)
\end{align*}
\textbf{Where:}
\begin{itemize}[leftmargin=*, noitemsep]
    \item $\Spot_0$: Initial spot exchange rate expressed as domestic currency per unit of foreign currency.
    \item $\Spot_t$: Expected spot exchange rate at future time horizon $t$.
    \item $i_L$: Expected inflation rate in the local (domestic/terms) country.
    \item $i_F$: Expected inflation rate in the foreign (base) country.
    \item $i_{L,k}, i_{F,k}$: Term-structure inflation rates prevailing during specific annual period $k$.
    \item $t$: Time horizon in years.
\end{itemize}
\end{formulabox}

\begin{formulabox}[The Fisher Effect \& Nominal Rate Decomposition]
\begin{align*}
    1 + r_{\text{nominal}} &= (1 + r_{\text{real}})(1 + i_{\text{inflation}}) \\[0.4em]
    r_{\text{nominal}} &\approx r_{\text{real}} + i_{\text{inflation}}
\end{align*}
\textbf{Where:}
\begin{itemize}[leftmargin=*, noitemsep]
    \item $r_{\text{nominal}}$: Nominal market interest rate quoted on debt contracts.
    \item $r_{\text{real}}$: Real rate of return required by investors in the absence of inflation.
    \item $i_{\text{inflation}}$: Anticipated or expected inflation rate over the contractual period.
    \item $\text{International Link}$: Real rates equalize across open capital markets ($r_{\text{real}, d} \approx r_{\text{real}, f}$), implying nominal interest differentials reflect relative inflation expectations ($i_d - i_f \approx \pi_d - \pi_f$).
\end{itemize}
\end{formulabox}

\begin{formulabox}[Covered Interest Rate Parity (CIP) \& Forward Margin Condition]
\begin{align*}
    \Forward &= \Spot \times \left( \frac{1 + i_d}{1 + i_f} \right) \\[0.5em]
    \frac{\Forward - \Spot}{\Spot} &\approx i_d - i_f
\end{align*}
\textbf{Where:}
\begin{itemize}[leftmargin=*, noitemsep]
    \item $\Forward$: Contractual forward exchange rate quoted outright.
    \item $\Spot$: Current spot exchange rate (direct quote: domestic currency per unit of foreign currency).
    \item $i_d$: Nominal interest rate prevailing in the domestic money market for tenor matching $\Forward$.
    \item $i_f$: Nominal interest rate prevailing in the foreign money market for tenor matching $\Forward$.
    \item $\frac{\Forward - \Spot}{\Spot}$: Annualized percentage forward premium ($>0$) or discount ($<0$) on the foreign currency.
    \item $\text{Arbitrage Condition}$: When $\Forward \neq \Spot \times \frac{1+i_d}{1+i_f}$, covered interest arbitrage generates riskless profit until market equilibrium is restored.
\end{itemize}
\end{formulabox}
