% 1. Definition Box
\newtcbtheorem[number within=section]{definition}{Definition}{
    enhanced,
    colback=lightgraybg,
    colframe=navyblue,
    coltitle=white,
    fonttitle=\bfseries\sffamily,
    attach boxed title to top left={yshift=-2mm, xshift=5mm},
    boxed title style={colback=navyblue, arc=1.5mm},
    separator sign={: },
    boxrule=1pt,
    arc=2mm,
    top=4mm,
    bottom=3mm,
    left=4mm,
    right=4mm,
    breakable
}{def}
