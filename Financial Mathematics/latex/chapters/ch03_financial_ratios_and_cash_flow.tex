\section{Fundamentals of Ratio Analysis \& Classification}
\lecturedate{12/10/2026}

\begin{definition}{Financial Ratio \& Ratio Analysis}{def_financial_ratio}
A \textbf{Financial Ratio} is a quantitative mathematical expression establishing a proportional, fractional, or percentage relationship between two or more interrelated accounting figures extracted from corporate balance sheets, income statements, or cash flow reports.

\textbf{Ratio Analysis} is the diagnostic technique of standardizing and interpreting these numerical proportions to assess the operational efficiency, liquidity, leverage, solvency, and profitability of a business concern.
\end{definition}

\subsection{Managerial Purposes \& Core Advantages of Ratio Analysis}
Financial statement ratios serve seven strategic corporate functions:
\begin{enumerate}[leftmargin=*, topsep=3pt, itemsep=3pt]
    \item \textbf{Financial Forecasting \& Budgeting:} Historical ratio trajectories reveal structural growth rates, operating expense ratios, and asset utilization rhythms, facilitating predictive multi-year pro forma cash flow modeling.
    \item \textbf{Managerial Control:} Acts as a continuous diagnostic monitoring instrument by highlighting negative variances between actual operational performance and benchmark targets.
    \item \textbf{Facilitating Corporate Communication:} Distills complex, voluminous accounting transactions into standardized, intuitive indicator scores readily comprehended by board members, institutional investors, and lenders.
    \item \textbf{Measuring Operational Efficiency:} Quantifies the rotation velocity and conversion efficiency with which assets and working capital are transformed into revenues.
    \item \textbf{Investment Decision Support:} Equips equity analysts and shareholders with normalized valuation multiples (e.g. $\text{P/E}$, dividend yields, return on capital) to identify mispriced securities.
    \item \textbf{Evaluating Financial Solvency:} Simultaneously examines short-term debt-servicing capacity (liquidity) and long-term capital structure stability (financial gearing and leverage).
    \item \textbf{Inter-Firm \& Intra-Firm Benchmarking:} Facilitates cross-sectional peer comparison against industry leaders as well as longitudinal trend analysis across consecutive operating cycles.
\end{enumerate}

\subsection{Profitability Ratios: Formulation \& Operational Margin Identities}
\textbf{Profitability Ratios} measure an enterprise's ability to generate operational surplus, gross margin, and net earnings relative to revenue, total assets, or equity investment:

\begin{formulabox}[Profitability Ratios Formulations]
\begin{itemize}[leftmargin=*, noitemsep]
    \item \textbf{Gross Profit Ratio:} Reflects margin after direct production costs ($\COGS$):
    \begin{equation}
        \text{Gross Profit Ratio} = \frac{\text{Gross Profit}}{\text{Net Sales}} \times 100
    \end{equation}
    where $\text{Net Sales} = \text{Total Sales} - \text{Sales Returns}$, and $\text{Gross Profit} = \text{Net Sales} - \COGS$.

    \item \textbf{Operating Profit Ratio:} Measures operational profit after administrative, selling, and depreciation expenses:
    \begin{equation}
        \text{Operating Profit Ratio} = \frac{\text{Operating Profit}}{\text{Net Sales}} \times 100
    \end{equation}
    where $\text{Operating Profit} = \text{Gross Profit} - \text{Operating Expenses} = \text{Net Operating Income}$.

    \item \textbf{Operating Ratio:} Measures aggregate cost efficiency (direct + operating expenses per rupee of sales):
    \begin{equation}
        \text{Operating Ratio} = \frac{\COGS + \text{Operating Expenses}}{\text{Net Sales}} \times 100
    \end{equation}

    \item \textbf{Fundamental Invariance Identity:}
    \begin{equation}
        \text{Operating Ratio} + \text{Operating Profit Ratio} = 100\%
    \end{equation}

    \item \textbf{Net Profit Ratio:} Final margin available to equity holders after non-operating expenses and taxes:
    \begin{equation}
        \text{Net Profit Ratio} = \frac{\text{Net Profit After Tax } (\NPAT)}{\text{Net Sales}} \times 100
    \end{equation}
\end{itemize}
\end{formulabox}

    \end{align*}
    \textit{Interpretation:} The enterprise retains $\inr 30.21$ from every $\inr 100$ of net revenue after absorbing direct manufacturing and procurement costs ($\COGS$).

    \item \textbf{Operating Profit Ratio:}
    \begin{align*}
        \text{Operating Profit Ratio} &= \frac{\text{Operating Profit}}{\text{Net Sales}} \times 100 = \frac{13{,}800}{96{,}000} \times 100 = \mathbf{14.38\%}
    \end{align*}
    \textit{Interpretation:} Pure core operational business activities yield an operating margin of $14.38\%$ before considering capital gains, dividend windfalls, or income taxes.

    \item \textbf{Operating Ratio:}
    \begin{align*}
        \COGS &= \text{Opening Stock} + \text{Purchases} - \text{Closing Stock} \\

    \end{align*}
    \textit{Verification of Invariance Identity:}
    \begin{equation*}
        \text{Operating Ratio} + \text{Operating Profit Ratio} = 85.63\% + 14.38\% = \mathbf{100.01\% \approx 100\%}
    \end{equation*}

    \item \textbf{Net Profit Ratio:}
    \begin{align*}
        \text{Net Profit Ratio} &= \frac{\text{Net Profit After Tax}}{\text{Net Sales}} \times 100 = \frac{8{,}300}{96{,}000} \times 100 = \mathbf{8.65\%}

    (If computed prior to non-operating adjustments as pure net operating income after tax: $\frac{13,800 - 7,000}{96,000} \times 100 = \frac{6,800}{96,000} \times 100 = 7.08\%$).
\end{enumerate}

\end{itemize}

\section{Turnover / Activity Ratios \& Operational Velocity}
\lecturedate{13/10/2026}

\begin{definition}{Turnover Ratio \& Operational Velocity}{def_turnover_ratio}
\textbf{Turnover Ratios} (also termed \textbf{Activity Ratios} or \textbf{Performance Ratios}) measure the operational efficiency and rotation velocity with which an enterprise converts its balance sheet assets and capital resources into revenue or cost of sales.

Unlike profitability metrics expressed as percentages, turnover ratios are mathematically expressed in \textbf{number of times} (rotation frequency per annum). A higher multiple generally signals superior managerial agility, efficient working capital deployment, and minimized idle capital.
\end{definition}

\begin{formulabox}[Comprehensive Activity and Velocity Formulations]
\begin{itemize}[leftmargin=*, noitemsep]
    \item \textbf{Inventory / Stock Turnover Ratio (ITR):}
    \begin{equation}
        \text{ITR} = \frac{\text{Cost of Goods Sold } (\COGS)}{\text{Average Inventory}} \quad \text{or} \quad \frac{\text{Net Sales}}{\text{Average Inventory}}
    \end{equation}
    where $\text{Average Inventory} = \dfrac{\text{Opening Stock} + \text{Closing Stock}}{2}$.
    \begin{equation}
        \text{Inventory Holding Period (Velocity)} = \frac{365}{\text{ITR}} \text{ days} = \frac{12}{\text{ITR}} \text{ months}
    \end{equation}

    \item \textbf{Debtors / Accounts Receivable Turnover Ratio (DTR):}
    \begin{equation}
        \text{DTR} = \frac{\text{Net Credit Sales}}{\text{Average Accounts Receivable}}
    \end{equation}
    where $\text{Net Credit Sales} = \text{Credit Sales} - \text{Sales Returns}$, and $\text{Average Receivables} = \text{Debtors} + \text{Bills Receivable}$.
    \begin{equation}
        \text{Debtors Collection Period (Velocity)} = \frac{365}{\text{DTR}} \text{ days} = \frac{12}{\text{DTR}} \text{ months}
    \end{equation}

    \item \textbf{Creditors / Accounts Payable Turnover Ratio (CTR):}
    \begin{equation}
        \text{CTR} = \frac{\text{Net Credit Purchases}}{\text{Average Accounts Payable}}
    \end{equation}
    where $\text{Net Credit Purchases} = \text{Credit Purchases} - \text{Purchase Returns}$, and $\text{Average Payables} = \text{Creditors} + \text{Bills Payable}$.
    \begin{equation}
        \text{Creditors Payment Period (Velocity)} = \frac{365}{\text{CTR}} \text{ days} = \frac{12}{\text{CTR}} \text{ months}
    \end{equation}

    \item \textbf{Working Capital Turnover Ratio:}
    \begin{equation}
        \text{Working Capital Turnover} = \frac{\text{Net Sales (or Cost of Sales)}}{\text{Net Working Capital}}
    \end{equation}
    where $\text{Net Working Capital} = \text{Current Assets} - \text{Current Liabilities}$.

    \item \textbf{Fixed Assets Turnover Ratio:}
    \begin{equation}
        \text{Fixed Assets Turnover} = \frac{\COGS}{\text{Net Fixed Assets}} \quad \text{or} \quad \frac{\text{Net Sales}}{\text{Net Fixed Assets}}
    \end{equation}

    \item \textbf{Capital Turnover Ratio:}
    \begin{equation}
        \text{Capital Turnover} = \frac{\text{Cost of Sales (or Net Sales)}}{\text{Capital Employed}}
    \end{equation}
\end{itemize}
\end{formulabox}

\lecturedate{13/10/2026}
\begin{example}{Comprehensive Calculation of Turnover Ratios and Velocity Periods}{ex_turnover_comprehensive}
\textbf{Problem:} \\
The accounting records of a manufacturing corporation disclose the following operating data for the financial year:
\begin{center}
\small
\begin{tabularx}{\linewidth}{>{\raggedright\arraybackslash}X r >{\raggedright\arraybackslash}X r}
\toprule
\textbf{Particulars} & \textbf{Amount (\inr)} & \textbf{Particulars} & \textbf{Amount (\inr)} \\
\midrule
Total Sales & 5,00,000 & Cash Purchases & 50,000 \\
Cash Sales & 40,000 & Purchase Returns & 10,000 \\
Sales Returns & 20,000 & Creditors at the end & 60,000 \\
Debtors at the end & 80,000 & Bills Payable at the end & 20,000 \\
Bills Receivable at the end & 20,000 & Reserve on Discount on Creditors & 2,000 \\
Reserve for Doubtful Debts & 1,000 & Opening Stock & 50,000 \\
Total Purchases & 3,00,000 & Closing Stock & 40,000 \\
Gross Profit & 1,00,000 & Fixed Assets & 10,00,000 \\
\bottomrule
\end{tabularx}
\end{center}
From the above information, compute:
\begin{enumerate*}[label=(\alph*), itemjoin=\quad]
    \item Stock Turnover Ratio,
    \item Fixed Asset Turnover Ratio (under both Cost of Sales and Net Sales bases),
    \item Debtors Turnover Ratio and Debtors Collection Period (in months and days),
    \item Creditors Turnover Ratio and Creditors Payment Period (in months and days).
\end{enumerate*}

\tcbline
\textbf{Reconciliation of Intermediate Financial Aggregates:}
\begin{enumerate}[leftmargin=*]
    \item \textbf{Net Sales:}
    \begin{equation*}
        \text{Net Sales} = \text{Total Sales} - \text{Sales Returns} = 5{,}00{,}000 - 20{,}000 = \mathbf{\inr 4{,}80{,}000}
    \end{equation*}

    \item \textbf{Cost of Goods Sold (Cost of Sales):}
    \begin{equation*}
        \COGS = \text{Net Sales} - \text{Gross Profit} = 4{,}80{,}000 - 1{,}00{,}000 = \mathbf{\inr 3{,}80{,}000}
    \end{equation*}

    \item \textbf{Average Stock:}
    \begin{equation*}
        \text{Average Stock} = \frac{\text{Opening Stock} + \text{Closing Stock}}{2} = \frac{50{,}000 + 40{,}000}{2} = \mathbf{\inr 45{,}000}
    \end{equation*}

    \item \textbf{Net Credit Sales:}
    \begin{align*}
        \text{Net Credit Sales} &= \text{Total Sales} - \text{Cash Sales} - \text{Sales Returns} \\
                                &= 5{,}00{,}000 - 40{,}000 - 20{,}000 = \mathbf{\inr 4{,}40{,}000}
    \end{align*}

    \item \textbf{Average Accounts Receivable:}
    (Note: Reserves for doubtful debts are provisions and are not deducted from gross receivables in turnover velocity analysis):
    \begin{equation*}
        \text{Accounts Receivable} = \text{Debtors} + \text{Bills Receivable} = 80{,}000 + 20{,}000 = \mathbf{\inr 1{,}00{,}000}
    \end{equation*}

    \item \textbf{Net Credit Purchases:}
    \begin{align*}
        \text{Net Credit Purchases} &= \text{Total Purchases} - \text{Cash Purchases} - \text{Purchase Returns} \\
                                    &= 3{,}00{,}000 - 50{,}000 - 10{,}000 = \mathbf{\inr 2{,}40{,}000}
    \end{align*}

    \item \textbf{Average Accounts Payable:}
    \begin{equation*}
        \text{Accounts Payable} = \text{Creditors} + \text{Bills Payable} = 60{,}000 + 20{,}000 = \mathbf{\inr 80{,}000}
    \end{equation*}
\end{enumerate}

\tcbline
\textbf{Step-by-Step Ratio Computations:}
\begin{enumerate}[leftmargin=*]
    \item \textbf{Stock Turnover Ratio:}
    \begin{equation*}
        \text{Stock Turnover Ratio} = \frac{\COGS}{\text{Average Stock}} = \frac{3{,}80{,}000}{45{,}000} = \mathbf{8.44 \text{ times}}
    \end{equation*}
    (If evaluated on Net Sales basis: $\frac{4,80,000}{45,000} = 10.67 \text{ times}$).

    \item \textbf{Fixed Asset Turnover Ratio:}
    \begin{itemize}[noitemsep]
        \item \textit{Based on Cost of Sales:}
        \begin{equation*}
            \text{Fixed Asset Turnover} = \frac{\COGS}{\text{Fixed Assets}} = \frac{3{,}80{,}000}{10{,}00{,}000} = \mathbf{0.38 \text{ times}}
        \end{equation*}
        \item \textit{Based on Net Sales:}
        \begin{equation*}
            \text{Fixed Asset Turnover} = \frac{\text{Net Sales}}{\text{Fixed Assets}} = \frac{4{,}80{,}000}{10{,}00{,}000} = \mathbf{0.48 \text{ times}}
        \end{equation*}
    \end{itemize}

    \item \textbf{Debtors Turnover Ratio and Debtors Collection Velocity:}
    \begin{equation*}
        \text{Debtors Turnover Ratio} = \frac{\text{Net Credit Sales}}{\text{Accounts Receivable}} = \frac{4{,}40{,}000}{1{,}00{,}000} = \mathbf{4.40 \text{ times}}
    \end{equation*}
    \begin{align*}
        \text{Collection Period (Months)} &= \frac{12}{\text{Debtors Turnover}} = \frac{12}{4.40} = \mathbf{2.73 \text{ months}} \\[0.4em]
        \text{Collection Period (Days)} &= \frac{365}{\text{Debtors Turnover}} = \frac{365}{4.40} = \mathbf{82.95 \approx 83 \text{ days}}
    \end{align*}

    \item \textbf{Creditors Turnover Ratio and Creditors Payment Velocity:}
    \begin{equation*}
        \text{Creditors Turnover Ratio} = \frac{\text{Net Credit Purchases}}{\text{Accounts Payable}} = \frac{2{,}40{,}000}{80{,}000} = \mathbf{3.00 \text{ times}}
    \end{equation*}
    \begin{align*}
        \text{Payment Period (Months)} &= \frac{12}{\text{Creditors Turnover}} = \frac{12}{3.00} = \mathbf{4.00 \text{ months}} \\[0.4em]
        \text{Payment Period (Days)} &= \frac{365}{\text{Creditors Turnover}} = \frac{365}{3.00} = \mathbf{121.67 \approx 122 \text{ days}}
    \end{align*}
\end{enumerate}

\textbf{Final Results Summary:}
\begin{itemize}[noitemsep]
    \item Stock Turnover Ratio = $\mathbf{8.44 \text{ times}}$
    \item Fixed Asset Turnover Ratio = $\mathbf{0.38 \text{ times}}$ (Cost of Sales) / $\mathbf{0.48 \text{ times}}$ (Net Sales)
    \item Debtors Turnover Ratio = $\mathbf{4.40 \text{ times}}$ ($\mathbf{2.73 \text{ months}}$ or $\mathbf{83 \text{ days}}$)
    \item Creditors Turnover Ratio = $\mathbf{3.00 \text{ times}}$ ($\mathbf{4.00 \text{ months}}$ or $\mathbf{122 \text{ days}}$)
\end{itemize}
\end{example}

\section{Solvency Ratios: Liquidity \& Financial Gearing}
\lecturedate{15/10/2026}

\begin{definition}{Solvency and Liquidity Diagnostic Framework}{def_solvency_liquidity}
\textbf{Solvency} denotes the overarching capacity of a business enterprise to fulfill its financial debt commitments as they mature. Solvency diagnostics operate across two interrelated temporal horizons:
\begin{enumerate}[leftmargin=*, topsep=2pt, itemsep=2pt]
    \item \textbf{Short-Term Solvency (Liquidity):} Assesses the immediate and near-term ability of a firm to settle its current debt commitments, trade obligations, and operating expenses without impairing ongoing business operations.
    \item \textbf{Long-Term Solvency (Financial Leverage):} Examines the structural adequacy of the firm's capital architecture to service long-term funded debt, periodic debenture interest charges, and preference dividend commitments over a multi-year horizon.
\end{enumerate}
\end{definition}

\begin{formulabox}[Short-Term Liquidity Ratios \& Regulatory Norms]
\begin{itemize}[leftmargin=*, noitemsep]
    \item \textbf{Current Ratio (Working Capital Ratio):}
    \begin{equation}
        \text{Current Ratio} = \frac{\text{Current Assets } (\text{CA})}{\text{Current Liabilities } (\text{CL})}
    \end{equation}
    where $\text{CA}$ denotes short-term resources (Inventories, Debtors, Cash, Bank, Bills Receivable, Prepaids, Marketable Securities), and $\text{CL}$ denotes short-term obligations (Creditors, Bills Payable, Bank Overdraft, Outstanding Expenses, Short-Term Provisions).
    \begin{equation*}
        \textbf{Conventional Ideal Benchmark Norm: } 2 : 1
    \end{equation*}

    \item \textbf{Quick Ratio / Liquid Ratio / Acid-Test Ratio:}
    Reflects the capacity to meet instant obligations without relying on inventory liquidations:
    \begin{equation}
        \text{Quick Ratio} = \frac{\text{Quick Assets } (\text{QA})}{\text{CL}} = \frac{\text{CA} - \text{Inventories} - \text{Prepaid Expenses}}{\text{Current Liabilities}}
    \end{equation}
    \begin{equation*}
        \textbf{Conventional Ideal Benchmark Norm: } 1 : 1
    \end{equation*}

    \item \textbf{Cash Position Ratio (Absolute Liquid Ratio):}
    The most rigorous measure of immediate cash liquidity, considering exclusively cash and near-cash balances:
    \begin{equation}
        \text{Cash Position Ratio} = \frac{\text{Cash} + \text{Bank} + \text{Current Marketable Investments}}{\text{Current Liabilities}}
    \end{equation}
    \begin{equation*}
        \textbf{Conventional Ideal Benchmark Norm: } 0.75 : 1 \quad (\text{or } 0.5 : 1)
    \end{equation*}
\end{itemize}
\end{formulabox}

\section{Fund Flow \& Cash Flow Statement Analysis}
\lecturedate{20/10/2026}

\begin{definition}{Financial Flow Statements: Funds Flow vs. Cash Flow}{def_financial_flow_statements}
Financial statement flow analysis examines the dynamic transmission of financial resources across consecutive accounting periods:
\begin{enumerate}[leftmargin=*, topsep=2pt, itemsep=2pt]
    \item \textbf{Funds Flow Statement:} A diagnostic statement portraying movements in the firm's \textbf{Net Working Capital} ($\NWC = \text{Current Assets} - \text{Current Liabilities}$). It explains how long-term resources (capital, debt borrowings, operational profits) were generated (Sources) and deployed into fixed capital or debt retirement (Applications).
    \item \textbf{Cash Flow Statement:} A specialized report portraying the gross inflows and outflows of \textbf{Cash and Cash Equivalents} (currency, demand deposits, short-term marketable instruments) arising from Operating, Investing, and Financing activities.
\end{enumerate}
\end{definition}

\subsection{Principles Governing Working Capital and Cash Movements}
The operational mechanics of working capital and cash adjustments follow complementary accounting principles:

\begin{center}
\small
\renewcommand{\arraystretch}{1.2}
\begin{tabularx}{\linewidth}{>{\raggedright\arraybackslash}X >{\centering\arraybackslash}p{4.2cm} >{\centering\arraybackslash}p{4.2cm}}
\toprule
\textbf{Accounting Transaction / Movement} & \textbf{Impact on Working Capital} & \textbf{Impact on Cash Position} \\
\midrule
Increase in Current Asset ($\uparrow \text{CA}$) & Increase in Working Capital ($+$) & Cash Outflow / Applied ($-$) \\
Decrease in Current Asset ($\downarrow \text{CA}$) & Decrease in Working Capital ($-$) & Cash Inflow / Released ($+$) \\
Increase in Current Liability ($\uparrow \text{CL}$) & Decrease in Working Capital ($-$) & Cash Inflow / Source ($+$) \\
Decrease in Current Liability ($\downarrow \text{CL}$) & Increase in Working Capital ($+$) & Cash Outflow / Applied ($-$) \\
\bottomrule
\end{tabularx}
\end{center}

\begin{formulabox}[Operational Cash Flow \& Funds Flow Identities]
\begin{itemize}[leftmargin=*, noitemsep]
    \item \textbf{Funds from Operations (FFO):}
    \begin{align}
        \FFO &= \text{Net Profit} + \text{Non-Cash Expenses (Depreciation, Amortization)} \notag \\
             &\quad + \text{Non-Operating Losses} - \text{Non-Operating Incomes}
    \end{align}

    \item \textbf{Cash Flow from Operations (CFO):}
    \begin{align}
        \CFO &= \FFO + \sum (\uparrow \text{CL}) + \sum (\downarrow \text{CA}) - \sum (\downarrow \text{CL}) - \sum (\uparrow \text{CA})
    \end{align}

    \item \textbf{Net Working Capital Identity:}
    \begin{equation}
        \Delta \NWC = \Delta \text{Current Assets} - \Delta \text{Current Liabilities}
    \end{equation}

    \item \textbf{Funds Flow Fundamental Balance Equation:}
    \begin{equation}
        \text{Total Sources of Funds} - \text{Total Applications of Funds} = \Delta \NWC
    \end{equation}
\end{itemize}
\end{formulabox}

\section{Cost-Volume-Profit (CVP) and Break-Even Analysis}
\lecturedate{26/10/2026}

\begin{definition}{Cost-Volume-Profit Analysis \& Marginal Costing}{def_cvp_analysis}
\textbf{Cost-Volume-Profit (CVP) Analysis} is a quantitative managerial economics technique that evaluates how dynamic variations in operating costs (both fixed and variable components) and production/sales volume interact to determine operational profit and economic feasibility.
\end{definition}

\subsection{Cost Architecture: Fixed vs. Variable Elements}
\begin{itemize}[leftmargin=*, topsep=2pt, itemsep=2pt]
    \item \textbf{Fixed Costs (Period Costs / Time Costs):} Operating expenditures that remain structurally invariant in total sum regardless of production volume fluctuations within the relevant operating capacity (e.g. factory rent, executive salaries, property taxes, straight-line depreciation). On a per-unit basis, fixed cost declines asymptotically as volume increases.
    \item \textbf{Variable Costs (Product Costs / Marginal Costs):} Operational expenditures that vary in direct linear proportion to aggregate output volume (e.g. direct raw materials, piece-rate direct labor, sales commissions, direct power). On a per-unit basis, variable cost remains constant.
\end{itemize}

\subsection{Fundamental CVP Formulations \& Identity Mechanics}
\begin{formulabox}[Comprehensive CVP and Break-Even Formulas]
\begin{itemize}[leftmargin=*, noitemsep]
    \item \textbf{Contribution Margin ($C$):}
    \begin{align}
        C &= \text{Sales Revenue } (S) - \text{Variable Cost } (V) \\[2pt]
        C &= \text{Fixed Cost } (F) + \text{Profit } (P) \\[2pt]
        C_{\text{unit}} &= s - v = \text{Selling Price per Unit} - \text{Variable Cost per Unit}
    \end{align}

    \item \textbf{Profit-Volume Ratio (P/V Ratio):}
    Expresses the percentage of sales revenue remaining after variable cost coverage:
    \begin{equation}
        \text{P/V Ratio} = \frac{C}{S} \times 100 = \frac{S - V}{S} \times 100 = \frac{F + P}{S} \times 100 = \frac{\Delta \text{Profit}}{\Delta \text{Sales}} \times 100
    \end{equation}

    \item \textbf{Break-Even Point (BEP):}
    The unique operational equilibrium volume where total sales revenue equals total operating costs ($\text{Total Profit} = 0$):
    \begin{align}
        \text{BEP (in Output Units)} &= \frac{\text{Fixed Cost } (F)}{\text{Contribution per Unit } (s - v)} = \frac{F}{C_{\text{unit}}} \\[4pt]
        \text{BEP (in Sales Value)} &= \frac{\text{Fixed Cost } (F)}{\text{P/V Ratio}} = \text{BEP (Units)} \times s
    \end{align}

    \item \textbf{Break-Even Ratio:}
    \begin{equation}
        \text{Break-Even Ratio} = \frac{\text{Break-Even Sales}}{\text{Actual / Budgeted Sales}} \times 100
    \end{equation}

    \item \textbf{Margin of Safety (MOS):}
    The cushion of actual sales volume exceeding the break-even threshold:
    \begin{align}
        \MOS &= \text{Actual Sales} - \text{Break-Even Sales} = \frac{\text{Profit}}{\text{P/V Ratio}} \\[3pt]
        \text{MOS Ratio} &= \frac{\MOS}{\text{Actual Sales}} \times 100 = 100\% - \text{Break-Even Ratio}
    \end{align}

    \item \textbf{Angle of Incidence ($\theta$):}
    The geometric angle formed at the intersection of the Total Revenue line ($y = s \cdot x$) and Total Cost line ($y = F + v \cdot x$) on the Break-Even Chart. A wider angle indicates a superior P/V ratio and accelerated profit growth beyond the BEP.
\end{itemize}
\end{formulabox}

\subsection{Managerial Advantages and Analytical Limitations}
\subsubsection{Strategic Advantages of Break-Even Analysis}
\begin{enumerate}[leftmargin=*, topsep=2pt, itemsep=2pt]
    \item \textbf{Cost Structure Delineation:} Precisely delineates total operating expenditure into fixed overheads and marginal components.
    \item \textbf{Threshold Volume Ascertainment:} Identifies the critical minimum sales output (in units and rupees) required to avert operational losses.
    \item \textbf{Decision-Making Optimization:} Facilitates pricing strategy, make-or-buy evaluations, and tender quoting through CVP simulation.
    \item \textbf{Inter-Firm Competitiveness Benchmarking:} Evaluates operating leverage and resilience against peer entities across business cycles.
    \item \textbf{Profit Forecasting \& Financial Planning:} Models target sales volume needed to generate pre-determined profit objectives: $S_{\text{target}} = \frac{F + P_{\text{target}}}{\text{P/V Ratio}}$.
    \item \textbf{Product Portfolio Optimization:} Evaluates multi-division product mixes to maximize aggregate contribution margins.
    \item \textbf{Cost Control \& Budgetary Benchmarking:} Establishes rigorous cost-absorption variance thresholds.
    \item \textbf{Corporate Surplus Aggregation:} Synthesizes divisional contributions to establish corporate earnings after central fixed overhead deduction.
\end{enumerate}

\subsubsection{Critical Theoretical \& Practical Limitations}
\begin{enumerate}[leftmargin=*, topsep=2pt, itemsep=2pt]
    \item \textbf{Imperfect Cost Segregation:} In practice, semi-variable and stepped-fixed costs fluctuate non-linearly, particularly beyond $100\%$ operating capacity.
    \item \textbf{Price Elasticity Violations:} Assumes a constant unit selling price regardless of volume, ignoring commercial price discounts and downward-sloping demand curves.
    \item \textbf{Detailed Cost Masking:} Aggregated break-even models obscure granular departmental efficiency variations.
    \item \textbf{Inventory Valuation Distortion:} Marginal costing ignores the absorption of fixed overheads into opening and closing stock reserves.
    \item \textbf{Multi-Product Mix Instability:} Assumes a rigid, constant product sales mix; shifts in relative consumer demand distort composite break-even coordinates.
    \item \textbf{Exclusion of Capital Structure Dynamics:} Fails to account for return on invested capital, debt leverage charges, and working capital carrying costs.
    \item \textbf{Static Environmental Assumptions:} Becomes highly unreliable under volatile macroeconomic conditions, input inflation, or technological disruption.
\end{enumerate}

\newpage
\lecturedate{26/10/2026}
\begin{example}{Break-Even Formulation and Visual Graphic Modeling for A Ltd}{ex_bep_a_ltd}
\textbf{Problem:} \\
A Ltd.~has furnished the following operating data:
\begin{itemize}[noitemsep]
    \item Fixed Cost ($F$) = $\inr 80{,}000$
    \item Variable Cost per unit ($v$) = $\inr 4$
    \item Estimated Sales Value ($S$) = $\inr 2{,}00{,}000$
    \item Selling Price per unit ($s$) = $\inr 20$
\end{itemize}
You are required to:
\begin{enumerate*}[label=(\alph*), itemjoin=\quad]
    \item Determine Contribution per unit and total,
    \item Calculate the P/V Ratio,
    \item Determine Break-Even Point in output units and monetary sales value,
    \item Calculate the Margin of Safety,
    \item Analyze the Angle of Incidence, and
    \item Construct the formal Break-Even Chart.
\end{enumerate*}

\tcbline
\textbf{Step 1: Intermediate Operational Calculations:}
\begin{enumerate}[leftmargin=*]
    \item \textbf{Estimated Output Volume:}
    \begin{equation*}
        \text{Output Units } (Q) = \frac{\text{Estimated Sales Value}}{\text{Selling Price per Unit}} = \frac{2{,}00{,}000}{20} = \mathbf{10{,}000 \text{ units}}
    \end{equation*}

    \item \textbf{Contribution Margin:}
    \begin{align*}
        C_{\text{unit}} &= s - v = 20 - 4 = \mathbf{\inr 16 \text{ per unit}} \\
        \text{Total Contribution} &= Q \times C_{\text{unit}} = 10{,}000 \times 16 = \mathbf{\inr 1{,}60{,}000}
    \end{align*}

    \item \textbf{Profit-Volume Ratio (P/V Ratio):}
    \begin{equation*}
        \text{P/V Ratio} = \frac{C_{\text{unit}}}{s} \times 100 = \frac{16}{20} \times 100 = \mathbf{80\%}
    \end{equation*}

    \item \textbf{Break-Even Point (BEP):}
    \begin{align*}
        \text{BEP (in Units)} &= \frac{\text{Fixed Cost}}{C_{\text{unit}}} = \frac{80{,}000}{16} = \mathbf{5{,}000 \text{ units}} \\[0.3em]
        \text{BEP (in Sales Value)} &= \frac{\text{Fixed Cost}}{\text{P/V Ratio}} = \frac{80{,}000}{0.80} = \mathbf{\inr 1{,}00{,}000} \\
        \text{Alternatively:} \quad \text{BEP (Value)} &= 5{,}000 \times 20 = \mathbf{\inr 1{,}00{,}000}
    \end{align*}

    \item \textbf{Margin of Safety (MOS):}
    \begin{align*}
        \MOS (\text{Value}) &= \text{Actual Sales} - \text{BEP (Sales)} \\
                            &= 2{,}00{,}000 - 1{,}00{,}000 = \mathbf{\inr 1{,}00{,}000} \\[0.3em]
        \MOS (\text{Units}) &= 10{,}000 - 5{,}000 = \mathbf{5{,}000 \text{ units}} \\[0.3em]
        \text{MOS Ratio} &= \frac{1{,}00{,}000}{2{,}00{,}000} \times 100 = \mathbf{50\%}
    \end{align*}

    \item \textbf{Angle of Incidence Analysis:}
    At the BEP ($5{,}000$ units, $\inr 1{,}00{,}000$), the sales revenue curve ($m_S = 20$) intersects the total cost curve ($m_{\text{TC}} = 4$). The slope differential $\Delta m = 20 - 4 = 16$ creates a steep angle of incidence ($\theta = \arctan(16)$), indicating aggressive profit expansion once operations cross $5{,}000$ units.
\end{enumerate}

\begin{example}{Contribution Break-Even Modeling and Sensitivity Analysis}{ex_contribution_bep}
\textbf{Problem:} \\
A manufacturing enterprise provides the following budgeted operational estimates:
\begin{itemize}[noitemsep]
    \item Budgeted Output Capacity = $80{,}000$ units
    \item Fixed Operating Expenses = $\inr 4{,}00{,}000$
    \item Variable Expenses per unit = $\inr 10$
    \item Selling Price per unit = $\inr 20$
\end{itemize}
You are required to:
\begin{enumerate*}[label=(\alph*), itemjoin=\quad]
    \item Calculate Contribution per unit and P/V Ratio,
    \item Determine Break-Even Point in units and in sales value,
    \item Compute Operating Profit at budgeted capacity,
    \item Construct the Contribution Break-Even Chart, and
    \item Conduct a Sensitivity Analysis for input cost and price fluctuations.
\end{enumerate*}

\tcbline
\textbf{Step 1: Fundamental Mathematical Computations:}
\begin{enumerate}[leftmargin=*]
    \item \textbf{Unit Contribution \& P/V Ratio:}
    \begin{align*}
        C_{\text{unit}} &= s - v = 20 - 10 = \mathbf{\inr 10 \text{ per unit}} \\
        \text{P/V Ratio} &= \frac{C_{\text{unit}}}{s} \times 100 = \frac{10}{20} \times 100 = \mathbf{50\%}
    \end{align*}

    \item \textbf{Break-Even Point (BEP):}
    \begin{align*}
        \text{BEP (in Units)} &= \frac{\text{Fixed Expenses}}{C_{\text{unit}}} = \frac{4{,}00{,}000}{10} = \mathbf{40{,}000 \text{ units}} \\[0.3em]
        \text{BEP (in Sales Value)} &= \frac{\text{Fixed Expenses}}{\text{P/V Ratio}} = \frac{4{,}00{,}000}{0.50} = \mathbf{\inr 8{,}00{,}000}
    \end{align*}

    \item \textbf{Operating Performance at Budgeted Output ($80{,}000$ units):}
    \begin{align*}
        \text{Gross Revenue } (80{,}000 \times \inr 20) &= \inr 16{,}00{,}000 \\
        \text{Less: Variable Expenses } (80{,}000 \times \inr 10) &= (\inr 8{,}00{,}000) \\
        \text{Total Contribution Margin} &= \inr 8{,}00{,}000 \\
        \text{Less: Fixed Operating Expenses} &= (\inr 4{,}00{,}000) \\
        \midrule
        \textbf{Budgeted Operating Profit} &= \mathbf{\inr 4{,}00{,}000}
    \end{align*}

    \item \textbf{Margin of Safety at Budgeted Output:}
    \begin{align*}
        \MOS (\text{Value}) &= 16{,}00{,}000 - 8{,}00{,}000 = \mathbf{\inr 8{,}00{,}000} \quad (40{,}000 \text{ units}) \\
        \text{MOS Ratio} &= \frac{8{,}00{,}000}{16{,}00{,}000} \times 100 = \mathbf{50\%}
    \end{align*}
\end{enumerate}

\textbf{Step 3: Sensitivity Analysis of Operational Variables:} \\
\textbf{Sensitivity Analysis} evaluates the quantitative responsiveness of profitability and break-even equilibrium to independent shifts in key operational parameters:
\begin{enumerate}[leftmargin=*, noitemsep, topsep=2pt]
    \item \textbf{Scenario A: 10\% Increase in Selling Price ($s \to \inr 22$):} \\
    $C_{\text{unit}} = 22 - 10 = \inr 12 \implies \text{BEP} = \frac{4{,}00{,}000}{12} = \mathbf{33{,}333 \text{ units}}$ ($\downarrow 16.7\%$). \\
    Operating profit at $80{,}000$ units surges to $(80{,}000 \times 12) - 4{,}00{,}000 = \mathbf{\inr 5{,}60{,}000}$ ($\uparrow 40\%$).

    \item \textbf{Scenario B: 10\% Increase in Variable Cost ($v \to \inr 11$):} \\
    $C_{\text{unit}} = 20 - 11 = \inr 9 \implies \text{BEP} = \frac{4{,}00{,}000}{9} = \mathbf{44{,}444 \text{ units}}$ ($\uparrow 11.1\%$). \\
    Operating profit at $80{,}000$ units declines to $(80{,}000 \times 9) - 4{,}00{,}000 = \mathbf{\inr 3{,}20{,}000}$ ($\downarrow 20\%$).

    \item \textbf{Scenario C: 10\% Increase in Fixed Overhead ($F \to \inr 4{,}40{,}000$):} \\
    $\text{BEP} = \frac{4{,}40{,}000}{10} = \mathbf{44{,}000 \text{ units}}$ ($\uparrow 10\%$). \\
    Operating profit declines to $8{,}00{,}000 - 4{,}40{,}000 = \mathbf{\inr 3{,}60{,}000}$ ($\downarrow 10\%$).
\end{enumerate}
\vspace{2pt}
\textit{Conclusion:} Profitability is most elastic with respect to unit selling price adjustments, followed by variable input cost control.
