\begin{formulabox}[{1. Simple Interest and Total Accumulated Amount}]
\begin{align*}
    \SI &= \frac{P \cdot N \cdot R}{100}, \qquad A = P + \SI = P \left(1 + \frac{N \cdot R}{100}\right)
\end{align*}
\textbf{Where:}
\begin{itemize}[noitemsep, topsep=1pt]
    \item $P$ = Principal sum borrowed or invested
    \item $N$ = Time period in years
    \item $R$ = Annual rate of interest ($\%$)
    \item $\SI$ = Simple interest earned
    \item $A$ = Total accumulated amount (Principal $+$ Interest)
\end{itemize}
\end{formulabox}

\begin{formulabox}[{2. Compound Interest and Multiple Conversion Periods}]
\begin{align*}
    \text{Annual Compounding Amount:} \quad A &= P \left(1 + \frac{R}{100}\right)^N \\[2pt]
    \text{Compound Interest Earned:} \quad \CI &= A - P = P \left[\left(1 + \frac{R}{100}\right)^N - 1\right] \\[2pt]
    \text{Sub-Annual Compounding ($m$ times/year):} \quad A &= P \left(1 + \frac{R}{m \cdot 100}\right)^{m \cdot N} \\[2pt]
    \text{Continuous Compounding:} \quad A &= P \cdot e^{r \cdot N} \qquad \left(r = \frac{R}{100}\right)
\end{align*}
\textbf{Where:}
\begin{itemize}[noitemsep, topsep=1pt]
    \item $P$ = Initial principal sum
    \item $R$ = Nominal annual rate of interest ($\%$)
    \item $r$ = Nominal interest rate in decimal form ($r = R/100$)
    \item $N$ = Total investment duration in years
    \item $m$ = Compounding frequency per annum ($2$ semi-annual, $4$ quarterly, $12$ monthly)
    \item $e$ = Base of the natural logarithm ($\approx 2.71828$)
    \item $A$ = Compound terminal amount
    \item $\CI$ = Compound interest earned
\end{itemize}
\end{formulabox}

\begin{formulabox}[{3. Effective Annual Rate of Interest (ERI / EAR) and Depreciation}]
\begin{align*}
    \text{Effective Annual Rate:} \quad \ERI &= \left(1 + \frac{R}{m \cdot 100}\right)^m - 1 \\[4pt]
    \text{Reducing Balance Depreciation:} \quad A &= P \left(1 - \frac{R}{100}\right)^N
\end{align*}
\textbf{Where:}
\begin{itemize}[noitemsep, topsep=2pt]
    \item $\ERI$ = Effective rate of interest (true annualized yield in decimal)
    \item $m$ = Number of compounding conversion periods per year
    \item $P$ = Original historical cost of the asset
    \item $R$ = Annual rate of depreciation ($\%$)
    \item $N$ = Operational lifespan / age in years
    \item $A$ = Written Down Value (WDV) or residual scrap value
\end{itemize}
\end{formulabox}

\begin{formulabox}[{4. Doubling Period Approximations}]
\begin{align*}
    \text{Exact Limit (Continuous):} \quad T_{\text{exact}} &= \frac{\ln 2}{r} \approx \frac{0.6931}{r} = \frac{69.3}{R} \\[4pt]
    \text{Rule of 72:} \quad T_{\text{Rule of 72}} &\approx \frac{72}{R} \\[4pt]
    \text{Rule of 69:} \quad T_{\text{Rule of 69}} &\approx 0.35 + \frac{69}{R}
\end{align*}
\textbf{Where:}
\begin{itemize}[noitemsep, topsep=2pt]
    \item $T$ = Number of years required for principal to double
    \item $R$ = Annual percentage interest rate ($R\%$)
    \item $r$ = Annual interest rate in decimal ($r = R/100$)
\end{itemize}
\end{formulabox}

\begin{formulabox}[{5. Annuity Cash Flows: Future Value and Present Value}]
\textbf{A. Closed-Form Factor Formulations:}
\begin{align*}
    \text{Ordinary Annuity Future Value:} \quad \FV &= A \cdot \CVAF(r, n) = A \left[ \frac{(1 + r)^n - 1}{r} \right] \\[4pt]
    \text{Annuity Due Future Value:} \quad \FV_{\text{due}} &= \FV_{\text{ord}} \times (1 + r) = A \left[ \frac{(1 + r)^n - 1}{r} \right] (1 + r) \\[4pt]
    \text{Present Value of a Lump Sum:} \quad \PV &= \frac{A}{(1 + r)^n} = A \cdot \PVF(r, n) \\[4pt]
    \text{Ordinary Annuity Present Value:} \quad \PV &= A \cdot \PVAF(r, n) = A \left[ \frac{1 - (1 + r)^{-n}}{r} \right] \\[4pt]
    \text{Annuity Due Present Value:} \quad \PV_{\text{due}} &= \PV_{\text{ord}} \times (1 + r) = A \left[ \frac{1 - (1 + r)^{-n}}{r} \right] (1 + r)
\end{align*}

\vspace{0.2em}
\textbf{B. Year-by-Year Summation Formulas ($\sum$):}
\begin{align*}
    \text{Ordinary Annuity FV:} \quad \FV &= \sum_{t=1}^{n} A \times (1 + r)^{n - t} \\[4pt]
    \text{Annuity Due FV:} \quad \FV_{\text{due}} &= \sum_{t=0}^{n-1} A \times (1 + r)^{n - t} \\[4pt]
    \text{Ordinary Annuity PV:} \quad \PV &= \sum_{t=1}^{n} \frac{A}{(1 + r)^t} = \frac{A}{(1 + r)^1} + \frac{A}{(1 + r)^2} + \dots + \frac{A}{(1 + r)^n} \\[4pt]
    \text{Annuity Due PV:} \quad \PV_{\text{due}} &= \sum_{t=0}^{n-1} \frac{A}{(1 + r)^t} = A + \frac{A}{(1 + r)^1} + \dots + \frac{A}{(1 + r)^{n-1}}
\end{align*}

\textbf{Where:}
\begin{itemize}[noitemsep, topsep=2pt]
    \item $A$ = Equal periodic cash payment / installment
    \item $r$ = Periodic interest or discount rate in decimal ($R/100$)
    \item $n$ = Total number of cash flow installments / conversion periods
    \item $t$ = Payment / discounting period index
    \item $\CVAF(r, n)$ = Compound Value Annuity Factor
    \item $\PVF(r, n)$ = Present Value Factor $\left((1 + r)^{-n}\right)$
    \item $\PVAF(r, n)$ = Present Value Annuity Factor
    \item $(1 + r)$ = Advance compounding factor for payments at beginning of period (annuity due)
\end{itemize}
\end{formulabox}

\begin{formulabox}[{6. Perpetuities, Finite Growing Annuity, and Sinking Funds}]
\textbf{A. Closed-Form Formulations:}
\begin{align*}
    \text{Constant Perpetuity:} \quad \PV &= \frac{C}{r} \\[4pt]
    \text{Growing Perpetuity:} \quad \PV &= \frac{C_1}{r - g} \qquad (r > g) \\[4pt]
    \text{Finite Growing Annuity:} \quad \PV &= \frac{C_1}{r - g} \left[ 1 - \left(\frac{1 + g}{1 + r}\right)^n \right] \qquad (r \ne g) \\[4pt]
    \text{Ordinary Sinking Fund Deposit:} \quad A &= \frac{\FV}{\CVAF(r, n)} = \FV \left[ \frac{r}{(1 + r)^n - 1} \right] \\[4pt]
    \text{Sinking Fund Annuity Due Deposit:} \quad A_{\text{due}} &= \frac{A_{\text{ord}}}{1 + r} = \frac{\FV}{\CVAF(r, n) \times (1 + r)}
\end{align*}

\vspace{0.2em}
\textbf{B. Year-by-Year Summation Formulation ($\sum$):}
\begin{align*}
    \text{Finite Growing Annuity PV:} \quad \PV &= \sum_{t=1}^{n} \frac{C_1 \times (1 + g)^{t-1}}{(1 + r)^t}
\end{align*}

\textbf{Where:}
\begin{itemize}[noitemsep, topsep=2pt]
    \item $C$ = Constant periodic perpetual cash inflow
    \item $C_1$ = Cash inflow generated at the end of Period 1
    \item $r$ = Periodic discount rate in decimal ($R/100$)
    \item $g$ = Constant compound growth rate of periodic cash flows ($G/100$)
    \item $n$ = Total number of payment periods
    \item $t$ = Cash flow period index ($t = 1, 2, \dots, n$)
    \item $\FV$ = Target reserve accumulation sum required at maturity
    \item $A$ = Periodic sinking fund contribution
    \item $\CVAF(r, n)$ = Compound Value Annuity Factor
\end{itemize}
\end{formulabox}

\begin{formulabox}[{7. Continuous Exponential Growth and Decay Models}]
\begin{align*}
    \text{Continuous Growth Model:} \quad n(t) &= n_0 \, e^{k t} \qquad (k > 0) \\[4pt]
    \text{Continuous Decay Model:} \quad n(t) &= n_0 \, e^{-k t} \qquad (k > 0) \\[4pt]
    \text{Rate Constant Calibration:} \quad k &= \frac{1}{T} \ln\left(\frac{n_{\max}}{n_{\min}}\right)
\end{align*}
\textbf{Where:}
\begin{itemize}[noitemsep, topsep=2pt]
    \item $n(t)$ = Modeled quantity / price impact level at elapsed time $t$
    \item $n_0$ = Initial quantity at baseline time $t = 0$
    \item $k$ = Continuous rate constant ($k > 0$)
    \item $T$ = Calibration interval between boundary values $[n_{\min}, n_{\max}]$
\end{itemize}
\end{formulabox}

\begin{formulabox}[{8. Security Yield Metrics and Holding Period Return}]
\begin{align*}
    \text{Current Yield} &= \frac{\text{Annual Dividend / Coupon}}{P_0} \times 100 \\[4pt]
    \text{Dividend Yield } (dY) &= \frac{\DPS}{P_0} \times 100 \\[4pt]
    \text{Holding Period Return } (\HPR) &= \frac{D_t + \Delta P}{P_1} = \frac{D_t + (P_t - P_1)}{P_1} \\[4pt]
    \text{Earnings Yield} &= \frac{\EPS}{P_0} \times 100
\end{align*}
\textbf{Where:}
\begin{itemize}[noitemsep, topsep=2pt]
    \item $P_0$ = Current prevailing market purchase price of the share/security
    \item $\DPS$ = Annual dividend per share
    \item $\HPR$ = Total rate of return realized over the investment horizon
    \item $D_t$ = Cash dividend received during the planned holding period
    \item $P_1$ = Initial purchase price paid at acquisition
    \item $P_t$ = Terminal market price realized at liquidation
    \item $\Delta P$ = Capital appreciation or depreciation ($P_t - P_1$)
    \item $\EPS$ = Earnings per share
\end{itemize}
\end{formulabox}

\begin{formulabox}[{9. General Dividend Discount Valuation Models (DDM)}]
\begin{align*}
    \text{Zero-Growth Model (Perpetual):} \quad P_0 &= \frac{D}{K_e} \\[4pt]
    \text{Finite 2-Year Holding Horizon:} \quad P_0 &= \frac{D_1}{1 + K_e} + \frac{D_2 + P_2}{(1 + K_e)^2} \\[4pt]
    \text{General Infinite DDM:} \quad P_0 &= \sum_{t=1}^{\infty} \frac{D_t}{(1 + K_e)^t} \\[4pt]
    \text{Gordon Constant Growth DDM:} \quad P_0 &= \frac{D_1}{K_e - g} = \frac{D_0 (1 + g)}{K_e - g} \qquad (K_e > g)
\end{align*}
\textbf{Where:}
\begin{itemize}[noitemsep, topsep=2pt]
    \item $P_0$ = Theoretical intrinsic market price per share today
    \item $D$ = Constant perpetual annual dividend per share
    \item $D_1, D_2, D_t$ = Expected future dividend per share at the end of year 1, 2, and $t$
    \item $D_0$ = Baseline dividend per share currently paid
    \item $P_2$ = Projected terminal market price at the end of Year 2
    \item $K_e$ = Required rate of return of equity investors / cost of equity capital
    \item $g$ = Constant perpetual annual compound growth rate in dividends ($g < K_e$)
\end{itemize}
\end{formulabox}

\begin{formulabox}[{10. Gordon's Dividend Capitalization Model with Endogenous Growth}]
\begin{align*}
    \text{Valuation Equation:} \quad P_0 &= \frac{\EPS_1 (1 - b)}{K_e - b \cdot r} = \frac{D_1}{K_e - g} \\[4pt]
    \text{Endogenous Growth Rate:} \quad g &= b \times r
\end{align*}
\textbf{Where:}
\begin{itemize}[noitemsep, topsep=2pt]
    \item $P_0$ = Theoretical market price per share ($\MPS$)
    \item $\EPS_1$ = Expected earnings per share for the upcoming period
    \item $b$ = Retention ratio (fraction of earnings retained within the firm, $0 \le b \le 1$)
    \item $(1 - b)$ = Dividend payout ratio ($D_1 / \EPS_1$)
    \item $K_e$ = Cost of equity / capitalization rate
    \item $r$ = Internal rate of return on reinvested corporate earnings (ROE)
    \item $g = b \cdot r$ = Endogenous growth rate of dividends and earnings
\end{itemize}
\vspace{0.4em}
\textbf{Payout Optimization Policy Rules:}
\begin{center}
\small
\begin{tabularx}{\linewidth}{l c c X}
    \toprule
    \textbf{Firm Category} & \textbf{Condition} & \textbf{Optimal Payout} & \textbf{Theoretical Dynamic} \\
    \midrule
    Growth Firm & $r > K_e$ & $\text{NIL } (0\%)$ & Maximum retention ($b=1$) maximizes MPS \\
    Normal Firm & $r = K_e$ & Indifferent & MPS is identical across all payout levels \\
    Declining Firm & $r < K_e$ & $100\%$ & Full distribution ($b=0$) maximizes MPS \\
    \bottomrule
\end{tabularx}
\end{center}
\end{formulabox}

\begin{formulabox}[{11. Walter's Valuation Formula and Firm Capitalization}]
\begin{align*}
    \text{Market Price Per Share:} \quad P &= \frac{D + \frac{r}{k}(E - D)}{k} \\[4pt]
    \text{Total Market Value of Firm:} \quad V &= N \times P
\end{align*}
\textbf{Where:}
\begin{itemize}[noitemsep, topsep=2pt]
    \item $P$ = Theoretical market price per share ($\MPS$)
    \item $D$ = Dividend per share paid ($\DPS = \EPS \times \text{Payout Ratio}$)
    \item $E$ = Earnings per share ($\EPS$)
    \item $(E - D)$ = Retained earnings per share
    \item $r$ = Internal rate of return on investment earned by the company
    \item $k$ = Cost of capital / capitalization rate
    \item $\frac{r}{k}(E - D)$ = Capitalized market value of the earnings retained
    \item $N$ = Total number of equity shares outstanding
    \item $V$ = Aggregate market valuation of the firm
\end{itemize}
\vspace{0.4em}
\textbf{Payout Optimization Policy Rules:}
\begin{center}
\small
\begin{tabularx}{\linewidth}{l c c X}
    \toprule
    \textbf{Firm Category} & \textbf{Condition} & \textbf{Optimum Payout} & \textbf{Share Price Dynamic} \\
    \midrule
    Growth Firm & $r > k$ & $0\%$ & MPS decreases as dividend payout increases \\
    Normal Firm & $r = k$ & Indifferent & MPS is invariant to payout choice \\
    Declining Firm & $r < k$ & $100\%$ & MPS increases as dividend payout increases \\
    \bottomrule
\end{tabularx}
\end{center}
\end{formulabox}

\begin{formulabox}[{12. Modigliani--Miller (MM) Dividend Irrelevance Model}]
\begin{align*}
    \text{Current Share Valuation:} \quad P_0 &= \frac{D_1 + P_1}{1 + K_e} \\[4pt]
    \text{Period-End Share Price:} \quad P_1 &= P_0 (1 + K_e) - D_1 \\[4pt]
    \text{External Financing Deficit:} \quad \Delta M &= I - (E - n \cdot D_1) \\[4pt]
    \text{New Shares to be Issued:} \quad m &= \Delta n = \frac{\Delta M}{P_1} = \frac{I - (E - n \cdot D_1)}{P_1} \\[4pt]
    \text{Total Market Value Invariant:} \quad V_1 &= (n + m) P_1 = (n + \Delta n) P_1 \\[4pt]
    &= n P_0 (1 + K_e) + I - E
\end{align*}
\textbf{Where:}
\begin{itemize}[noitemsep, topsep=2pt]
    \item $P_0$ = Current prevailing market price per share at beginning of period
    \item $P_1$ = Market price per share at the end of the period
    \item $D_1$ = Dividend per share declared/distributed at the end of the period
    \item $K_e$ = Rate of capitalization / cost of equity capital appropriate to risk class
    \item $I$ = Total capital investment budget required for the period
    \item $E$ = Total net income (earnings) of the firm during the period
    \item $n$ = Number of existing equity shares outstanding at beginning of period
    \item $n D_1$ = Aggregate dividend distribution to existing shareholders
    \item $\Delta M$ = External equity financing deficit to be raised
    \item $m = \Delta n$ = Number of newly issued equity shares
    \item $V_1$ = Total aggregate market valuation of the firm at year end
\end{itemize}
\end{formulabox}

\begin{formulabox}[{13. Bond Valuation, Yield to Maturity (YTM), and Yield to Call (YTC)}]
\textbf{A. Theoretical Present Value Valuation:}
\begin{align*}
    V_0 &= \sum_{t=1}^{n} \frac{C}{(1 + k_d)^t} + \frac{M}{(1 + k_d)^n} = C \cdot \PVAF(k_d, n) + M \cdot \PVF(k_d, n) \\[4pt]
    V_{\text{call}} &= \sum_{t=1}^{N_c} \frac{C}{(1 + k_c)^t} + \frac{CP}{(1 + k_c)^{N_c}} = C \cdot \PVAF(k_c, N_c) + CP \cdot \PVF(k_c, N_c)
\end{align*}

\vspace{0.2em}
\textbf{B. Approximation Formulas:}
\begin{align*}
    \text{Approximate YTM} &= \frac{C + \dfrac{M - P_0}{n}}{\dfrac{M + P_0}{2}} \\[6pt]
    \text{Approximate YTC} &= \frac{C + \dfrac{CP - P_0}{N_c}}{\dfrac{CP + P_0}{2}}
\end{align*}

\textbf{Where:}
\begin{itemize}[noitemsep, topsep=2pt]
    \item $V_0$ = Theoretical intrinsic economic value of the bond
    \item $P_0$ = Current prevailing market purchase price of the bond
    \item $C$ = Annual coupon interest payment ($C = M \times \text{Coupon Rate}$)
    \item $M$ = Par or face value of the bond payable at final maturity
    \item $n$ = Total number of years until final stated maturity
    \item $N_c$ = Number of years remaining until the earliest call date
    \item $CP$ = Call price payable upon redemption ($CP = M + \text{Call Premium}$)
    \item $k_d$ = Periodic required rate of return / discount rate of bondholders
    \item $k_c$ = True Yield to Call rate ($\YTC$)
    \item $\PVAF(k, n)$ = Present Value Annuity Factor at rate $k$ for $n$ periods
    \item $\PVF(k, n)$ = Present Value Factor at rate $k$ for $n$ periods
\end{itemize}
\end{formulabox}

\begin{formulabox}[{14. Loan Amortization and Equal Periodic Installment (EMI)}]
\begin{align*}
    \text{Periodic Loan Installment (EMI):} \quad E &= \frac{P \cdot r \cdot (1 + r)^n}{(1 + r)^n - 1} = \frac{P}{\PVAF(r, n)} \\[4pt]
    \text{Interest Component for Period } t: \quad I_t &= B_{t-1} \times r \\[4pt]
    \text{Principal Amortization for Period } t: \quad \Delta P_t &= E - I_t \\[4pt]
    \text{Closing Principal Balance for Period } t: \quad B_t &= B_{t-1} - \Delta P_t = B_{t-1} - (E - I_t)
\end{align*}
\textbf{Where:}
\begin{itemize}[noitemsep, topsep=2pt]
    \item $E$ = Equal periodic loan installment / repayment (EMI)
    \item $P = B_0$ = Initial principal loan amount borrowed
    \item $r$ = Periodic interest rate in decimal ($R / (m \cdot 100)$)
    \item $n$ = Total number of periodic repayment installments ($N \times m$)
    \item $t$ = Amortization period index ($t = 1, 2, \dots, n$)
    \item $B_{t-1}$ = Outstanding principal balance at the beginning of period $t$
    \item $B_t$ = Outstanding principal balance remaining at the end of period $t$
    \item $I_t$ = Interest charge incurred during period $t$
    \item $\Delta P_t$ = Principal reduction portion of installment $E$ in period $t$
    \item $\PVAF(r, n)$ = Present Value Annuity Factor for $n$ periods at rate $r$
\end{itemize}
\end{formulabox}

\begin{formulabox}[{15. Price--Earnings (P/E) Valuation Model and Earnings Per Share}]
\begin{align*}
    \text{Earnings Per Share (EPS):} \quad \EPS &= \frac{\EAT - D_p}{N} \\[4pt]
    \text{Price--Earnings Ratio (P/E):} \quad \text{P/E Ratio} &= \frac{P_0}{\EPS} \\[4pt]
    \text{Implied Stock Valuation:} \quad P_0 &= \EPS \times (\text{P/E Ratio}) \\[4pt]
    \text{Theoretical P/E Multiple (from Gordon DDM):} \quad \frac{P_0}{\EPS_1} &= \frac{1 - b}{K_e - g} = \frac{\text{Payout Ratio}}{K_e - g}
\end{align*}
\textbf{Where:}
\begin{itemize}[noitemsep, topsep=2pt]
    \item $\EPS$ = Earnings per share
    \item $\EAT$ = Corporate earnings after tax (net income)
    \item $D_p$ = Total preference dividend obligations payable ($\EAT - D_p$ = Earnings to Common Shareholders)
    \item $N$ = Weighted average number of common equity shares outstanding
    \item $P_0$ = Current market price per share ($\MPS$)
    \item $\text{P/E Ratio}$ = Price-to-earnings valuation multiple
    \item $\EPS_1$ = Projected forward earnings per share for the next annual period
    \item $b$ = Retention ratio of earnings ($0 \le b \le 1$)
    \item $(1 - b)$ = Dividend payout ratio ($D_1 / \EPS_1$)
    \item $K_e$ = Cost of equity / required rate of return
    \item $g$ = Sustainable constant annual growth rate in earnings and dividends ($g = b \cdot r$)
\end{itemize}
\end{formulabox}

\begin{formulabox}[{16. Profitability Ratios and Operational Margin Identities}]
\begin{align*}
    \text{Gross Profit Ratio} &= \frac{\text{Gross Profit}}{\text{Net Sales}} \times 100 \\[4pt]
    \text{Operating Profit Ratio} &= \frac{\text{Operating Profit}}{\text{Net Sales}} \times 100 \\[4pt]
    \text{Operating Ratio} &= \frac{\COGS + \text{Operating Expenses}}{\text{Net Sales}} \times 100 \\[4pt]
    \text{Invariance Identity:} &\quad \text{Operating Ratio} + \text{Operating Profit Ratio} = 100\% \\[4pt]
    \text{Net Profit Ratio} &= \frac{\NPAT}{\text{Net Sales}} \times 100 \\[4pt]
    \text{Return on Investment (ROI):} \quad \text{ROI} &= \frac{\text{Operating Profit (EBIT)}}{\text{Capital Employed}} \times 100
\end{align*}
\textbf{Where:}
\begin{itemize}[noitemsep, topsep=2pt]
    \item $\text{Net Sales}$ = Total gross sales revenue less sales returns and allowances
    \item $\COGS$ = Cost of Goods Sold ($\text{Opening Stock} + \text{Purchases} + \text{Direct Expenses} - \text{Closing Stock}$)
    \item $\text{Gross Profit}$ = Net revenue remaining after direct manufacturing/trading costs ($\text{Net Sales} - \COGS$)
    \item $\text{Operating Expenses}$ = Administrative, office, selling, distribution, and depreciation expenses
    \item $\text{Operating Profit}$ = Core operational earnings before non-operating income, interest, and tax ($\text{EBIT}$)
    \item $\NPAT$ = Final net income available after non-operating gains/losses and corporate income taxes
    \item $\text{Capital Employed}$ = Total long-term funds deployed ($\text{Shareholders' Equity} + \text{Long-Term Debt}$)
\end{itemize}
\end{formulabox}

\begin{formulabox}[{17. Activity and Turnover Ratios with Operational Velocity Periods}]
\begin{align*}
    \text{Inventory Turnover Ratio (ITR):} \quad \text{ITR} &= \frac{\COGS}{\text{Average Inventory}} \quad \text{or} \quad \frac{\text{Net Sales}}{\text{Average Inventory}} \\[4pt]
    \text{Inventory Holding Velocity:} \quad T_{\text{inventory}} &= \frac{365}{\text{ITR}} \text{ days} = \frac{12}{\text{ITR}} \text{ months} \\[4pt]
    \text{Debtors Turnover Ratio (DTR):} \quad \text{DTR} &= \frac{\text{Net Credit Sales}}{\text{Average Accounts Receivable}} \\[4pt]
    \text{Debtors Collection Velocity:} \quad T_{\text{debtors}} &= \frac{365}{\text{DTR}} \text{ days} = \frac{12}{\text{DTR}} \text{ months} \\[4pt]
    \text{Creditors Turnover Ratio (CTR):} \quad \text{CTR} &= \frac{\text{Net Credit Purchases}}{\text{Average Accounts Payable}} \\[4pt]
    \text{Creditors Payment Velocity:} \quad T_{\text{creditors}} &= \frac{365}{\text{CTR}} \text{ days} = \frac{12}{\text{CTR}} \text{ months} \\[4pt]
    \text{Fixed Assets Turnover:} \quad \text{FATR} &= \frac{\COGS \text{ (or Net Sales)}}{\text{Net Fixed Assets}} \\[4pt]
    \text{Working Capital Turnover:} \quad \text{WCTR} &= \frac{\text{Cost of Sales (or Net Sales)}}{\text{Net Working Capital}}
\end{align*}
\textbf{Where:}
\begin{itemize}[noitemsep, topsep=2pt]
    \item $\text{Average Inventory}$ = $\frac{1}{2}(\text{Opening Inventory} + \text{Closing Inventory})$
    \item $\text{Net Credit Sales}$ = Total gross sales less cash sales and sales returns
    \item $\text{Average Accounts Receivable}$ = Sundry Debtors $+$ Bills Receivable (gross before doubtful debt reserves)
    \item $\text{Net Credit Purchases}$ = Total gross purchases less cash purchases and purchase returns
    \item $\text{Average Accounts Payable}$ = Sundry Creditors $+$ Bills Payable
    \item $\text{Net Fixed Assets}$ = Book value of property, plant, and equipment less accumulated depreciation
    \item $\text{Net Working Capital}$ = $\text{Current Assets} - \text{Current Liabilities}$
    \item $T_{\text{inventory}}, T_{\text{debtors}}, T_{\text{creditors}}$ = Operational velocity turnover periods in calendar days or months
\end{itemize}
\end{formulabox}
