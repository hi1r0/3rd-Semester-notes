% 1. Definition Box (Clean black border with light gray tinted interior)
\newtcbtheorem[number within=section]{definition}{Definition}{
    enhanced,
    colback=boxfill,
    colframe=pureblack,
    coltitle=white,
    fonttitle=\bfseries\sffamily\small,
    attach boxed title to top left={yshift=-2mm, xshift=5mm},
    boxed title style={colback=pureblack, arc=1mm},
    minipage boxed title=\linewidth-1.5cm,
    separator sign={: },
    boxrule=0.9pt,
    arc=1.5mm,
    top=4mm,
    bottom=3mm,
    left=4mm,
    right=4mm,
    breakable
}{def}

    colback=boxfill,
    colframe=pureblack,
    title=#1,
    fonttitle=\bfseries\sffamily\scriptsize,
    coltitle=white,
    attach boxed title to top left={yshift=-1.5mm, xshift=3mm},

    top=2mm,
    bottom=2mm,
    left=3mm,
    right=3mm,
