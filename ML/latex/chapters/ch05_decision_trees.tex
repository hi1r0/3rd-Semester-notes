\chapter{Decision Trees}

\section{Overview and Core Concepts}

A \textbf{Decision Tree} is a supervised machine learning model that predicts a target outcome by recursively splitting the data into smaller, progressively more homogeneous (pure) subsets based on feature values.

\subsection{Structural Components of a Decision Tree}
\begin{itemize}
    \item \textbf{Root Node:} The topmost node of the tree, containing all training samples prior to any partitioning.
    \item \textbf{Decision Node (Internal Node):} A node that tests a specific feature condition to split data into child branches.
    \item \textbf{Branch:} A directed link representing the outcome of a decision test leading to a subsequent node.
    \item \textbf{Leaf Node (Terminal Node):} An end node carrying the final prediction (class label in classification or mean value in regression); it undergoes no further splitting.
    \item \textbf{Parent Node:} A node immediately preceding a split.
    \item \textbf{Child Node:} A node generated immediately as the outcome of a parent node split.
    \item \textbf{Tree Depth:} The maximum number of levels (edges) from the root node to the deepest leaf node.
\end{itemize}

\subsection{General Recursive Splitting Algorithm}
\begin{enumerate}
    \item \textbf{Initialization:} Place all training observations at the root node.
    \item \textbf{Check Termination:} If a stopping condition is met (e.g., node is pure, maximum depth reached, or insufficient samples), convert the node into a terminal leaf.
    \item \textbf{Evaluate Candidate Splits:} For every available feature and potential split point, compute the impurity reduction across the subsets.
    \item \textbf{Select Optimal Split:} Select the feature and threshold that yields the best split criterion:
    \begin{itemize}
        \item Maximum Information Gain (ID3 algorithm),
        \item Maximum Gain Ratio (C4.5 algorithm),
        \item Maximum Gini Gain / Minimum Weighted Gini Impurity (CART algorithm), or
        \item Maximum Variance Reduction (for continuous regression trees).
    \end{itemize}
    \item \textbf{Recursive Partitioning:} Partition the sample observations into disjoint child subsets and recurse on each child node.
    \item \textbf{Stopping Criteria:} Terminate tree growth when:
    \begin{itemize}
        \item The node is completely pure (all samples share the same class),
        \item The predefined maximum depth has been reached,
        \item The minimum sample size required for a split is breached, or
        \item No available candidate split yields a positive impurity improvement.
    \end{itemize}
\end{enumerate}

\section{Impurity Measures and Splitting Criteria}

\subsection{Standard Tree Induction Algorithms}
Different decision tree induction algorithms utilize distinct impurity metrics and optimization objectives:

\begin{center}\renewcommand{\arraystretch}{1.35}
\begin{tabular}{lll}
\toprule
\textbf{Algorithm} & \textbf{Impurity Measure} & \textbf{Split Criterion} \\
\midrule
\textbf{ID3} & Entropy & Information Gain \\
\textbf{C4.5} & Entropy & Gain Ratio \\
\textbf{CART} & Gini Impurity & Gini Gain / Minimum Weighted Gini \\
\bottomrule
\end{tabular}
\end{center}

\subsection{Entropy and Information Gain}
\textbf{Entropy} quantifies the amount of uncertainty or disorder in a dataset $S$ containing $c$ classes:
\begin{equation}
H(S) = -\sum_{i=1}^{c} p_i \log_2(p_i)
\end{equation}
where $p_i = \frac{\text{Number of samples in class } i}{\text{Total observations in node}}$ is the empirical class probability.

For a two-class problem ($c=2$ with probabilities $p_1$ and $p_2$):
\begin{equation}
H(S) = -p_1 \log_2(p_1) - p_2 \log_2(p_2)
\end{equation}
\begin{itemize}
    \item A node containing only one class is completely pure: $H(S) = 0$.
    \item Maximum entropy occurs under equal class balance ($p_1 = p_2 = 0.5 \implies H(S) = 1.0$).
\end{itemize}

\textbf{Information Gain (IG)} measures the reduction in entropy achieved by partitioning a parent dataset $S$ using attribute $A$:
\begin{equation}
IG(S, A) = H(\text{parent}) - H(\text{children}) = H(S) - \sum_{v \in \text{Values}(A)} \frac{|S_v|}{|S|} H(S_v)
\end{equation}
where $H(\text{children}) = \sum_i \frac{n_i}{n} H_i$ represents the weighted average entropy across all resulting child partitions $S_v$ of size $n_i$.

\subsection{Gain Ratio (C4.5)}
To overcome ID3's bias toward attributes with many distinct values, the \textbf{C4.5} algorithm normalizes Information Gain by the split information:
\begin{equation}
\text{Gain Ratio}(S, A) = \frac{IG(S, A)}{\text{Split Info}(S, A)}
\end{equation}
where $\text{Split Info}(S, A) = -\sum_{v \in \text{Values}(A)} \frac{|S_v|}{|S|} \log_2\left(\frac{|S_v|}{|S|}\right)$.

\subsection{Gini Impurity (CART)}
\textbf{Gini Impurity} measures the probability of misclassifying a randomly chosen element from the set if it were randomly labeled according to the class distribution:
\begin{equation}
\text{Gini}(S) = 1 - \sum_{i=1}^{c} p_i^2
\end{equation}
For a binary classification setting ($c = 2$):
\begin{equation}
\text{Gini}(S) = 1 - p_1^2 - p_2^2
\end{equation}
\begin{itemize}
    \item \textbf{Smaller Gini:} Indicates a purer node.
    \item \textbf{Larger Gini:} Indicates a more mixed, impure node.
    \item $\text{Gini} = 0 \implies$ The node is completely pure (all samples belong to a single class).
    \item For binary classification, maximum Gini is $0.5$ when $p_1 = p_2 = 0.5$.
\end{itemize}

\subsection{Comparison: Gini Impurity vs. Entropy}

\begin{center}\renewcommand{\arraystretch}{1.35}
\begin{tabular}{lll}
\toprule
\textbf{Property} & \textbf{Gini Impurity} & \textbf{Entropy} \\
\midrule
\textbf{Measures} & Impurity & Uncertainty \\
\textbf{Minimum Value} & $0$ & $0$ \\
\textbf{Pure Node Value} & $0$ & $0$ \\
\textbf{Binary Maximum} & $0.5$ (at $p = 0.5$) & $1.0$ (at $p = 0.5$) \\
\textbf{Good Split Condition} & Lower Weighted Gini & Higher Information Gain \\
\textbf{Computation} & Simpler (algebraic squares) & Heavy (evaluates logarithms) \\
\textbf{Common Usage} & CART & ID3 / C4.5 algorithms \\
\bottomrule
\end{tabular}
\end{center}

\subsection{Advantages and Disadvantages of Decision Trees}

\begin{center}\renewcommand{\arraystretch}{1.35}
\begin{tabular}{p{0.46\textwidth} p{0.48\textwidth}}
\toprule
\textbf{Advantages} & \textbf{Disadvantages} \\
\midrule
\begin{itemize}[leftmargin=*, nosep]
    \item Highly intuitive and easy to understand.
    \item Easy to visualize graphically.
    \item Easy to explain to non-technical audiences.
    \item Captures complex non-linear relationships.
    \item Directly applicable to classification and regression.
\end{itemize}
&
\begin{itemize}[leftmargin=*, nosep]
    \item Highly susceptible to overfitting on training data.
    \item Deep trees become unwieldy and overly complex.
    \item High variance: small data perturbations alter the tree.
    \item A single tree has limited predictive stability.
\end{itemize} \\
\bottomrule
\end{tabular}
\end{center}
