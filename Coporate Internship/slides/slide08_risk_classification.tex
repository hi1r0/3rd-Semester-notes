        \begin{column}{0.48\textwidth}
            \begin{mathcard}[height=5.4cm]{Composite Scoring \& Marginal Drag}
                \fontsize{6.3pt}{8pt}\selectfont
                Hierarchical weighted aggregation across all 5 analytical pillars:
                \begin{equation*}
                    \text{Score} = \sum_{k=1}^5 W_k \left( \sum_{i \in \text{Pillar}_k} w_{k,i}' \cdot S(x_i) \right)
                \end{equation*}
                where $\sum W_k = 1.00$ ($30\%$ Profit, $25\%$ Solv, $15\%$ Grow, $15\%$ Liq, $15\%$ Eff).

                \vspace{3pt}
                \textbf{\color{deepemerald}Ratio Contribution Analysis ($\Delta S_i$):}
                \begin{equation*}
                    \Delta S_i = W_k \cdot w_{k,i}' \times \big(S(x_i) - 50.0\big)
                \end{equation*}
                \begin{itemize}
                    \setlength{\itemsep}{1.5pt}
                    \item $\Delta S_i > 0$: Net positive alpha contributor above peer median.
                    \item $\Delta S_i < 0$: Specific metric creating point drag on composite health.
                \end{itemize}

                \vspace{2pt}
                \textbf{\color{deepemerald}Quartile Positioning Tags:}
                \begin{itemize}
                    \setlength{\itemsep}{1pt}
                    \item $x_i \ge Q_3$: \badge[jade]{Opportunity}
                    \item $Q_1 \le x_i < Q_3$: \badge[slatygray]{Neutral}
                    \item $x_i < Q_1$: \badge[forestgreen]{Risk Watch}
                \end{itemize}
                \vfill
                \badge[jade]{Hierarchical Aggregation}
            \end{mathcard}
        \end{column}
