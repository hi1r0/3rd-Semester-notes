\begin{frame}{Analytical Motivations \& Screening Latency}{Line-Item Variance and Asynchronous Disclosures Inhibit Scalable Equity Screening}
    \vspace{-0.20cm}
    \begin{columns}[T,onlytextwidth]
        % Column 1: Financial Reporting Bottlenecks
        \begin{column}{0.485\textwidth}
            \begin{quantcard}[height=2.74cm]{1. Schema Heterogeneity \& Alias Variance}
                \fontsize{5.8pt}{7.0pt}\selectfont
                \begin{itemize}
                    \setlength{\itemsep}{1.5pt}
                    \item \textbf{Alias Heterogeneity:} Inconsistent canonical line-item naming in public filings (\textit{Revenue from Operations} vs.\ \textit{Total Revenue} vs.\ \textit{Gross Sales}).
                    \item \textbf{Statement Fragmentation:} Discrepant structures across standalone versus consolidated financial disclosures.
                    \item \textbf{Fiscal Period Asynchrony:} Staggered quarterly filing dates delay robust cross-sectional peer grouping.
                \end{itemize}
            \end{quantcard}

            \vspace{0.12cm}

            \begin{quantcard}[height=2.74cm]{2. Manual Screening Scalability Limits}
                \fontsize{5.8pt}{7.0pt}\selectfont
                \begin{itemize}
                    \setlength{\itemsep}{1.5pt}
                    \item \textbf{Audit Fatigue:} Manual auditing across 2,000+ listed NSE equities is computationally prohibitive and prone to transcription errors.
                    \item \textbf{Earnings Season Surge:} Analyst bottlenecks during quarterly disclosures delay critical credit warnings and rating downgrades.
                    \item \textbf{Brittle Models:} Ad-hoc spreadsheet models crash on missing rows or zero-denominator singularities.
                \end{itemize}
            \end{quantcard}
        \end{column}

        % Column 2: Systemic Consequences & Quantitative Objectives
        \begin{column}{0.485\textwidth}
            \begin{quantcard}[height=2.74cm]{3. Latent Insolvency Detection Lag}
                \fontsize{5.8pt}{7.0pt}\selectfont
                \begin{itemize}
                    \setlength{\itemsep}{1.5pt}
                    \item \textbf{Silent Health Erosion:} Working capital deficits and cash-flow decay precede rating agency downgrades by 2 to 3 quarters.
                    \item \textbf{Decoupled Heuristics:} Conventional screening fails to correlate high accounting margins with deteriorating cash conversion.
                    \item \textbf{Information Asymmetry:} Retail and institutional desks lack immediate peer-relative ranking signals.
                \end{itemize}
            \end{quantcard}

            \vspace{0.12cm}

            \begin{quantcard}[height=2.74cm]{4. Production Engineering Objectives}
                \fontsize{5.8pt}{7.0pt}\selectfont
                \begin{itemize}
                    \setlength{\itemsep}{1.5pt}
                    \item \textbf{Autonomous Ingestion:} Automated retrieval and caching of multi-year balance sheets and cash flows.
                    \item \textbf{Deterministic Normalization:} Unify discordant aliases into 16 standardized fundamental financial ratios.
                    \item \textbf{Sub-2-Second SLA:} Deliver instantaneous cross-sectional peer diagnostics without manual spreadsheet intervention.
                \end{itemize}
            \end{quantcard}
        \end{column}
    \end{columns}
\end{frame}
