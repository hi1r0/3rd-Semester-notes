        \begin{column}{0.485\textwidth}
            \begin{quantcard}[height=2.7cm]{1. Schema Heterogeneity \& Latency}
                \fontsize{6.5pt}{8.3pt}\selectfont
                \textbf{\color{deepemerald}Line-Item Variance Across NSE Sectors}
                \begin{itemize}
                    \setlength{\itemsep}{1pt}
                    \item Inconsistent canonical aliases in Yahoo Finance filings.
                    \item Missing quarterly disclosures and asynchronous fiscal years.
                \end{itemize}
                \textbf{\color{deepemerald}Manual Screening Prohibitive}
                \begin{itemize}
                    \setlength{\itemsep}{1pt}
                    \item Manual auditing of 2,000+ equities invites fatigue and errors.
                \end{itemize}
            \end{quantcard}

            \vspace{0.1cm}

            \begin{quantcard}[height=2.75cm]{2. Gaussian Mean Distortion}
                \fontsize{6.5pt}{8.3pt}\selectfont
                Classical $Z$-score collapses on fat-tailed financial distributions:
                \begin{equation*}
                    Z_i = \frac{x_i - \bar{\mu}}{\hat{\sigma}}, \quad \text{Breakdown Point } = 0.0\%
                \end{equation*}
                \begin{itemize}
                    \setlength{\itemsep}{1pt}
                    \item A single distressed entity inflates sample variance $\hat{\sigma}^2$ and pulls sample mean $\bar{\mu}$.
                    \item \textbf{Masking Effect:} Moderate peer distress appears artificially "normal".
                \end{itemize}
            \end{quantcard}
        \end{column}
