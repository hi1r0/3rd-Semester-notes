            \begin{mathcard}[height=2.74cm]{Tukey Fences Formulation}
                \fontsize{5.7pt}{6.9pt}\selectfont
                Evaluates empirical spread of the central $50\%$ peer cohort:
                \begin{equation*}
                    \textstyle \mathcal{F}_{\text{lower}} = Q_1 - 1.5 \cdot \IQR, \quad \mathcal{F}_{\text{upper}} = Q_3 + 1.5 \cdot \IQR
                \end{equation*}
                \begin{equation*}
                    \textstyle \text{Outlier}(x_i) = \mathbf{1}_{\{x_i < \mathcal{F}_{\text{lower}} \;\lor\; x_i > \mathcal{F}_{\text{upper}}\}}
                \end{equation*}
                \begin{itemize}
                    \setlength{\itemsep}{1.2pt}
                    \item \textbf{Mild Anomaly ($1.5\,\text{IQR}$):} Flags candidate peer deviation.
                    \item \textbf{Extreme Anomaly ($3.0\,\text{IQR}$):} Structural reporting divergence.
                \end{itemize}
            \end{mathcard}

            \vspace{0.12cm}

            \begin{quantcard}[height=2.74cm]{Non-Parametric Invariants \& Properties}
                \fontsize{5.8pt}{7.0pt}\selectfont
                \begin{itemize}
                    \setlength{\itemsep}{1.5pt}
                    \item \textbf{25\% Breakdown Resistance:} Up to $25\%$ of observations in each tail can be distorted without shifting quartile anchors $Q_1$ or $Q_3$.
                    \item \textbf{Asymmetry Preservation:} Adapts to skewed metrics like Debt/Equity and P/E ratios without false penalties.
                    \item \textbf{Distribution-Free:} Robust against heavy-tailed financial shocks.
                \end{itemize}
            \end{quantcard}
