            \begin{mathcard}[height=2.65cm]{Tukey Fences Formulation}
                \fontsize{6.3pt}{8.2pt}\selectfont
                Evaluates empirical spread of the central $50\%$ peer cohort:
                \begin{equation*}
                    \mathcal{F}_{\text{lower}} = Q_1 - 1.5 \times \IQR, \quad \mathcal{F}_{\text{upper}} = Q_3 + 1.5 \times \IQR
                \end{equation*}
                \begin{equation*}
                    \text{Outlier}(x_i) = \mathbf{1}_{\{x_i < \mathcal{F}_{\text{lower}} \;\lor\; x_i > \mathcal{F}_{\text{upper}}\}}
                \end{equation*}
                \textbf{\color{deepemerald}Two-Tier Severity Thresholds:}
                \begin{itemize}
                    \setlength{\itemsep}{1.5pt}
                    \item \textbf{Mild Anomaly ($1.5\,\text{IQR}$):} Flags candidate peer deviation.
                    \item \textbf{Extreme Anomaly ($3.0\,\text{IQR}$):} Structural reporting divergence.
                \end{itemize}
            \end{mathcard}

            \vspace{0.15cm}

            \begin{quantcard}[height=2.65cm]{Non-Parametric Invariants \& Properties}
                \fontsize{6.4pt}{8.3pt}\selectfont
                \textbf{\color{deepemerald}1. 25\% Breakdown Resistance}
                \begin{itemize}
                    \setlength{\itemsep}{1.5pt}
                    \item Up to $25\%$ of observations in each tail can be infinitely distorted without shifting quartile anchors $Q_1$ or $Q_3$.
                \end{itemize}
                \textbf{\color{deepemerald}2. Asymmetry \& Skew Preservation}
                \begin{itemize}
                    \setlength{\itemsep}{1.5pt}
                    \item Unlike symmetric $(\bar{\mu} \pm 2\hat{\sigma})$, Tukey fences naturally adapt to skewed distributions like P/E ratios and Debt/Equity.
                    \item Prevents penalizing firms operating in intrinsically skewed industries.
                \end{itemize}
            \end{quantcard}
