            \begin{mathcard}[height=2.7cm]{Tukey Fences Cutoff Bounds}
                \fontsize{6.3pt}{8pt}\selectfont
                Evaluates empirical spread of central $50\%$ peer mass:
                \begin{equation*}
                    \mathcal{F}_{\text{lower}} = Q_1 - 1.5 \times \IQR, \quad \mathcal{F}_{\text{upper}} = Q_3 + 1.5 \times \IQR
                \end{equation*}
                \begin{equation*}
                    \text{Outlier}(x_i) = \mathbf{1}_{\{x_i < \mathcal{F}_{\text{lower}} \;\lor\; x_i > \mathcal{F}_{\text{upper}}\}}
                \end{equation*}
                Incorruptible by extreme fat tails in the upper or lower 25\%.
            \end{mathcard}

            \vspace{0.1cm}

            \begin{mathcard}[height=2.75cm]{Modified MAD Z-Score}
                \fontsize{6.1pt}{7.7pt}\selectfont
                Calculates median dispersion relative to peer median:
                \begin{equation*}
                    \MAD = \median\Big(\big|x_i - \median(X)\big|\Big)
                \end{equation*}
                \begin{equation*}
                    M_i = \frac{0.6745 \cdot |x_i - \median(X)|}{\MAD}, \quad M_i > 3.5 \implies \text{Outlier}
                \end{equation*}
                \textbf{\color{deepemerald}Why 0.6745?} Under $X \sim \mathcal{N}$, $\mathbb{E}[\MAD] = \Phi^{-1}(0.75)\,\sigma \approx 0.6745\,\sigma$, guaranteeing standard Gaussian consistency with a \textbf{50\% breakdown point}.
            \end{mathcard}
