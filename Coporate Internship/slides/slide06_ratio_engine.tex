            \begin{mathcard}[height=2.74cm]{Mathematical Guardrail: safe\_divide}
                \fontsize{5.7pt}{6.9pt}\selectfont
                Eliminates catastrophic division-by-zero singularities:
                \begin{equation*}
                    \textstyle \safedivide(a, b, \epsilon) = 
                    \begin{cases} 
                        \frac{a}{b} & \text{if } |b| > \epsilon \\[1pt]
                        \text{NaN} & \text{if } |b| \le \epsilon \text{ or } b = \text{NaN}
                    \end{cases}
                \end{equation*}
                \begin{itemize}
                    \setlength{\itemsep}{1.2pt}
                    \item \textbf{Machine Threshold:} $\epsilon = 10^{-9}$ protects zero-interest debt-free firms.
                    \item \textbf{Graceful Propagation:} Missing items yield typed NaNs for dynamic weights.
                \end{itemize}
            \end{mathcard}
