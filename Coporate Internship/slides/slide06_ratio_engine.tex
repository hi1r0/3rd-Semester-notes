                \fontsize{6.3pt}{8.2pt}\selectfont
                Eliminates catastrophic division-by-zero singularities:
                \begin{equation*}
                    \safedivide(a, b, \epsilon) = 
                    \begin{cases} 
                        \dfrac{a}{b} & \text{if } |b| > \epsilon \\[3pt]
                        \text{NaN} & \text{if } |b| \le \epsilon \text{ or } b = \text{NaN}
                    \end{cases}
                \end{equation*}
                \textbf{\color{deepemerald}Numerical Stability Invariant:}
                \begin{itemize}
                    \setlength{\itemsep}{1.5pt}
                    \item Setting machine threshold $\epsilon = 10^{-9}$ protects against zero interest expense in debt-free firms without halting batch execution.
                    \item Missing line items gracefully yield typed IEEE NaNs for downstream dynamic weight redistribution.
                \end{itemize}
            \end{mathcard}
