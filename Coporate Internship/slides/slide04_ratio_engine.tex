                \fontsize{6.3pt}{8pt}\selectfont
                Eliminates floating-point division-by-zero singularities:
                \begin{equation*}
                    \safedivide(a, b, \epsilon) = 
                    \begin{cases} 
                        \dfrac{a}{b} & \text{if } |b| > \epsilon \\[3pt]
                        \text{NaN} & \text{if } |b| \le \epsilon \text{ or } b = \text{NaN}
                    \end{cases}
                \end{equation*}
                Threshold $\epsilon = 10^{-9}$ preserves numerical robustness when zero interest expense or negligible debt is recorded.
            \end{mathcard}
