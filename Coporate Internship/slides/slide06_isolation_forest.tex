        % Column 1: Algorithmic Mathematics
        \begin{column}{0.51\textwidth}
            \begin{mathcard}[height=5.4cm]{Multivariate Tree Isolation Mechanics}
                \fontsize{6.3pt}{7.9pt}\selectfont
                Recursively partitions the 16-ratio space $\mathbb{R}^{16}$ via random hyperplanes.
                Anomalies require fewer cuts to isolate near the tree root.

                \vspace{2pt}
                \textbf{\color{deepemerald}Average Path Length of Unsuccessful Search in BST:}
                \begin{equation*}
                    c(n) = 2\ln(n - 1) + 2\gamma - \frac{2(n - 1)}{n}
                \end{equation*}
                where $\gamma \approx 0.5772156649$ is the Euler--Mascheroni constant and $n$ is the peer sub-sampling size ($n=256$).

                \vspace{2pt}
                \textbf{\color{deepemerald}Anomaly Score Formulation:}
                \begin{equation*}
                    s(x, n) = 2^{-\frac{\mathbb{E}(h(x))}{c(n)}}
                \end{equation*}
                \begin{itemize}
                    \setlength{\itemsep}{1.5pt}
                    \item $\mathbb{E}(h(x)) \to 0 \implies s \to 1$: Highly isolated outlier.
                    \item $\mathbb{E}(h(x)) \to c(n) \implies s \to 0.5$: Structurally typical peer.
                    \item Classification Rule: $s(x, n) \ge 0.60 \implies \text{Outlier}$.
                \end{itemize}
                \vfill
                \badge[jade]{Multivariate Tree Ensemble}
            \end{mathcard}
        \end{column}
