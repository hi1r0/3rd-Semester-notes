        % Column 1: Mathematical Derivation & 0.6745 Factor
        \begin{column}{0.485\textwidth}
            \begin{mathcard}[height=2.74cm]{Mathematical Derivation: Why 0.6745?}
                \fontsize{5.7pt}{6.9pt}\selectfont
                Median Absolute Deviation relative to peer sample median:
                \begin{equation*}
                    \textstyle \MAD = \median\Big(\big|x_i - \median(X)\big|\Big)
                \end{equation*}
                \textbf{\color{deepemerald}Normal Consistency Proof:}
                \begin{itemize}
                    \setlength{\itemsep}{1pt}
                    \item For $X \sim \mathcal{N}(\mu, \sigma^2)$, $|X - \mu|/\sigma$ has CDF $F(z) = 2\Phi(z) - 1$.
                    \item Setting $F(z) = 0.50 \implies \Phi(z) = 0.75 \implies z \approx 0.6745$.
                    \item Thus, $\mathbb{E}[\MAD] = \Phi^{-1}(0.75)\,\sigma \approx 0.6745\,\sigma$.
                \end{itemize}
            \end{mathcard}

            \vspace{0.12cm}

            \begin{mathcard}[height=2.74cm]{Asymptotically Unbiased Scale Estimator}
                \fontsize{5.7pt}{6.9pt}\selectfont
                Inverting the expectation yields robust standard deviation $\hat{\sigma}_{\MAD}$:
                \begin{equation*}
                    \textstyle \hat{\sigma}_{\MAD} = \frac{\MAD}{\Phi^{-1}(0.75)} \approx 1.4826 \times \MAD
                \end{equation*}
                \textbf{\color{deepemerald}Modified Z-Score Formulation:}
                \begin{equation*}
                    \textstyle M_i = \frac{|x_i - \median(X)|}{\hat{\sigma}_{\MAD}} = \frac{0.6745 \cdot |x_i - \median(X)|}{\MAD}
                \end{equation*}
                \begin{itemize}
                    \setlength{\itemsep}{0.8pt}
                    \item Retains the intuitive scale of classical $Z$, immune to masking.
                \end{itemize}
            \end{mathcard}
        \end{column}

        % Column 2: Decision Rules & 50% Breakdown Point
        \begin{column}{0.485\textwidth}
            \begin{quantcard}[height=2.74cm]{Decision Rule \& Outlier Classification}
                \fontsize{5.8pt}{7.0pt}\selectfont
                \textbf{\color{deepemerald}Iglewicz \& Hoaglin (1993) Criterion}
                \begin{equation*}
                    \textstyle \text{Outlier}(x_i) = \mathbf{1}_{\{M_i > 3.5\}}
                \end{equation*}
                \begin{itemize}
                    \setlength{\itemsep}{1.2pt}
                    \item $M_i > 3.5$ represents a $3.5\sigma$ deviation relative to robust dispersion, isolating genuine structural irregularities.
                    \item Sample size $n$ does not dilute the anomaly score of distressed peers.
                \end{itemize}
                \vfill
                \badge[jade]{Iglewicz-Hoaglin Robust Metric Benchmark}

            \begin{quantcard}[height=2.74cm]{Maximal 50\% Breakdown Point Invariant}
                \fontsize{5.8pt}{7.0pt}\selectfont
                \textbf{\color{deepemerald}Theoretical Upper Bound of Robustness}
                \begin{equation*}
                    \textstyle \varepsilon^*(\MAD) = \frac{\lfloor n/2 \rfloor}{n} \xrightarrow{n \to \infty} 50.0\%

                \begin{itemize}
                    \setlength{\itemsep}{1.2pt}
                    \item Up to \textbf{half the sector cohort} can be driven to arbitrary $\pm\infty$ extremes without causing $\median(X)$ or $\MAD$ to collapse.
                    \item Guarantees flawless anomaly isolation during cyclical shocks.
                \end{itemize}
            \end{quantcard}
