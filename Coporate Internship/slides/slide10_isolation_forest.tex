        \begin{column}{0.485\textwidth}
            \begin{mathcard}[height=2.74cm]{Path-Length Expectation $c(n)$}
                \fontsize{5.7pt}{6.9pt}\selectfont
                Average unsuccessful search depth in a Binary Search Tree (BST):
                \begin{equation*}
                    \textstyle c(n) = 2\ln(n - 1) + 2\gamma - \frac{2(n - 1)}{n}, \quad \gamma \approx 0.5772
                \end{equation*}
                \begin{itemize}
                    \setlength{\itemsep}{0.8pt}
                    \item Provides an equivalent null-model benchmark for random recursive data partitioning across $n$ samples.
                    \item Serves as normalizing divisor for path depth $\mathbb{E}[h(x)]$.
                \end{itemize}
            \end{mathcard}

            \vspace{0.12cm}

            \begin{mathcard}[height=2.74cm]{Exponential Anomaly Scoring Function}
                \fontsize{5.5pt}{6.7pt}\selectfont
                Monotonic mapping of tree depth into $[0, 1]$ anomaly score:
                \begin{equation*}
                    \textstyle s(x, n) = 2^{-\mathbb{E}[h(x)]/c(n)}, \quad s \ge 0.60 \implies \text{Outlier}
                \end{equation*}
                \textbf{\color{deepemerald}Asymptotic Bounds:}
                \begin{itemize}
                    \setlength{\itemsep}{0.3pt}
                    \item $\mathbb{E}[h(x)] \to 0 \implies s \to 1.0$ (definite structural anomaly).
                    \item $\mathbb{E}[h(x)] \to c(n) \implies s \to 0.5$ (typical peer behaviour).
                    \item $\mathbb{E}[h(x)] \to n - 1 \implies s \to 0.0$ (dense core cluster).
                \end{itemize}
            \end{mathcard}
        \end{column}
