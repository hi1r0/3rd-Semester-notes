                \begin{itemize}
                    \setlength{\itemsep}{2pt}

                    \item Classical univariate filters cannot detect joint coordinate decoupling across 16 financial dimensions.
                \end{itemize}

                \begin{itemize}
                    \setlength{\itemsep}{1.5pt}
                    \item Subsampling isolates anomalies while eliminating swamping hazards.

                \begin{itemize}
                    \setlength{\itemsep}{2pt}
                    \item \textbf{Tier 3 (Critical Alert):} Concurrently flagged by Tukey IQR, MAD $Z$-score, and Isolation Forest (unanimous distress signal).
                    \item \textbf{Tier 2 (Elevated Watch):} Flagged by 2 of 3 detectors (subtle balance sheet overhang or accrual-cash decoupling).

        % Column 2: Mathematical Scoring Function & Asymptotics
        \begin{column}{0.485\textwidth}
            \begin{mathcard}[height=2.65cm]{Path-Length Expectation $c(n)$}
                \fontsize{6.2pt}{8.0pt}\selectfont
                Average unsuccessful search depth in a Binary Search Tree (BST):
                \begin{equation*}
                    c(n) = 2\ln(n - 1) + 2\gamma - \frac{2(n - 1)}{n}, \quad \gamma \approx 0.5772
                \end{equation*}
                \begin{itemize}
                    \setlength{\itemsep}{1.5pt}
                    \item Provides an equivalent null-model benchmark for random recursive data partitioning across $n$ samples.
                    \item Serves as the normalizing divisor for path depth $\mathbb{E}[h(x)]$.
                \end{itemize}
            \end{mathcard}

            \vspace{0.15cm}

            \begin{mathcard}[height=2.65cm]{Exponential Anomaly Scoring Function}
                \fontsize{6.2pt}{8.0pt}\selectfont
                Monotonic mapping of tree depth into $[0, 1]$ anomaly score:
                \begin{equation*}
                    s(x, n) = 2^{-\frac{\mathbb{E}[h(x)]}{c(n)}}, \quad s(x, n) \ge 0.60 \implies \text{Outlier}
                \end{equation*}
                \textbf{\color{deepemerald}Asymptotic Bounds:}
                \begin{itemize}
                    \setlength{\itemsep}{1.5pt}
                    \item $\mathbb{E}[h(x)] \to 0 \implies s \to 1.0$ (definite structural anomaly).
                    \item $\mathbb{E}[h(x)] \to c(n) \implies s \to 0.5$ (typical peer behaviour).
                    \item $\mathbb{E}[h(x)] \to n - 1 \implies s \to 0.0$ (dense core cluster).
                \end{itemize}
            \end{mathcard}
        \end{column}
    \end{columns}
