\begin{frame}{Gaussian Breakdown \& Outlier Masking}{Parametric Mean Collapse Under Heavy-Tailed Financial Distributions}
    \vspace{-0.20cm}
    \begin{columns}[T,onlytextwidth]
        % Column 1: Theoretical Breakdown & Proof of Masking
        \begin{column}{0.485\textwidth}
            \begin{mathcard}[height=2.74cm]{Parametric Gaussian Vulnerability}
                \fontsize{5.7pt}{6.9pt}\selectfont
                Classical $Z$-score collapses on fat-tailed financial distributions:
                \begin{equation*}
                    \textstyle Z_i = \dfrac{x_i - \bar{\mu}}{\hat{\sigma}}, \quad \text{Breakdown Point } \varepsilon^* = 0.0\%
                \end{equation*}
                \begin{itemize}
                    \setlength{\itemsep}{1.2pt}
                    \item \textbf{Zero Breakdown Invariant:} $\varepsilon^* = \min\{m/n : \sup |T(\mathcal{X}')| = \infty\}$. A single outlier $x_0 \to \infty$ pulls $\bar{\mu} \to \infty$ and $\hat{\sigma}^2 \to \infty$.
                    \item \textbf{Heavy Tail Invalidation:} Pareto-like equity ratios violate Gaussian normality.
                \end{itemize}
            \end{mathcard}

            \vspace{0.12cm}

            \begin{mathcard}[height=2.74cm]{Mathematical Proof: Masking \& Swamping}
                \fontsize{5.7pt}{6.9pt}\selectfont
                Sample $\{x_1, \dots, x_{n-1}\}$ with single severe anomaly $x_0 \gg 0$:
                \begin{equation*}
                    \textstyle \bar{\mu} \approx \frac{x_0}{n}, \quad \hat{\sigma} \approx \frac{x_0}{\sqrt{n}} \implies Z_k = \frac{x_k - \bar{\mu}}{\hat{\sigma}} \approx -\frac{1}{\sqrt{n}}
                \end{equation*}
                \begin{itemize}
                    \setlength{\itemsep}{1.2pt}
                    \item \textbf{Masking Hazard:} As $n$ grows, $Z_k \to 0$. Truly distressed firms remain nestled safely within $[-2, +2]$, evading detection.
                    \item \textbf{Swamping Distortion:} Inflated $\hat{\sigma}$ falsely flags healthy borderline peers.
                \end{itemize}
            \end{mathcard}
        \end{column}
