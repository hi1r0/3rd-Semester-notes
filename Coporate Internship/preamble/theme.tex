% Executive Dark Mode Palette (Zero Gold)
\definecolor{paperbg}{HTML}{071411}       % #071411 Deep obsidian emerald canvas
\definecolor{cardbg}{HTML}{0D211B}        % #0D211B Elevated dark card container
\definecolor{cardinner}{HTML}{132F26}     % #132F26 Inset code & formula box
\definecolor{mintborder}{HTML}{1E4E3F}    % #1E4E3F Subtle dark emerald border outline
\definecolor{deepemerald}{HTML}{34D399}   % #34D399 Radiant mint for titles & headers
\definecolor{forestgreen}{HTML}{10B981}   % #10B981 Glowing jade for cards & subtitles
\definecolor{jade}{HTML}{059669}          % #059669 Vibrant accent for active nodes
\definecolor{darkcharcoal}{HTML}{F8FAFC}  % #F8FAFC Platinum white body text
\definecolor{slatygray}{HTML}{94A3B8}     % #94A3B8 Cool silver for secondary text
\definecolor{minttint}{HTML}{132F26}      % #132F26 Inset tint
\definecolor{lightsage}{HTML}{0D211B}     % Fallback to cardbg

% Apply Global Beamer Colors
\setbeamercolor{background canvas}{bg=paperbg}
\setbeamercolor{normal text}{fg=darkcharcoal}
\setbeamercolor{structure}{fg=deepemerald}
\setbeamercolor{alerted text}{fg=jade}
\setbeamercolor{example text}{fg=forestgreen}
\setbeamercolor{frametitle}{bg=paperbg, fg=deepemerald}
\setbeamercolor{framesubtitle}{fg=slatygray}
\setbeamercolor{itemize item}{fg=forestgreen}
\setbeamercolor{itemize subitem}{fg=slatygray}
\setbeamercolor{itemize subsubitem}{fg=slatygray}

% Custom Bullets (No Triangles - Radiant Jade Micro-Dots & Rings)
\setbeamertemplate{itemize item}{\tikz[baseline=-0.5ex]{\draw[fill=forestgreen, draw=none] (0,0) circle (1.8pt);}}
\setbeamertemplate{itemize subitem}{\tikz[baseline=-0.5ex]{\draw[draw=slatygray, line width=0.6pt, fill=none] (0,0) circle (1.4pt);}}
\setbeamertemplate{itemize subsubitem}{\tikz[baseline=-0.5ex]{\draw[fill=slatygray, draw=none] (0,0) circle (1.0pt);}}

% Remove default navigation symbols
\setbeamertemplate{navigation symbols}{}

% Page margins for 16:9 widescreen
\setbeamersize{text margin left=0.75cm, text margin right=0.75cm}

\setbeamertemplate{headline}{}

% Custom Frametitle Banner
\setbeamertemplate{frametitle}{%
    \nointerlineskip%
    \begin{beamercolorbox}[wd=\paperwidth, leftskip=0.75cm, rightskip=0.75cm]{frametitle}%
        \vspace{0.20cm}%
        {\fontsize{12.5pt}{14.5pt}\selectfont\bfseries\color{deepemerald}\insertframetitle}\par%
        \ifx\insertframesubtitle\@empty\else%
            \vspace{2pt}%
            {\fontsize{7.5pt}{9.0pt}\selectfont\color{slatygray}\raggedright\textsf{\insertframesubtitle}}\par%
        \fi%
        \vspace{0.10cm}%
        {\color{mintborder}\rule{\dimexpr\paperwidth-1.5cm\relax}{0.7pt}}\par%
    \end{beamercolorbox}%
}

% Completely Empty Footline (Per User Directive)
\setbeamertemplate{footline}{}
